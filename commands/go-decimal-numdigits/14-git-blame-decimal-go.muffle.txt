192	// NewFromString returns a new Decimal from a string representation.
193	// Trailing zeroes are not trimmed.
194	//
195	// Example:
196	//
197	//	d, err := NewFromString("-123.45")
198	//	d2, err := NewFromString(".0001")
199	//	d3, err := NewFromString("1.47000")
200	func NewFromString(value string) (Decimal, error) {
201		originalInput := value
202		var intString string
203		var exp int64
204	
205		// Check if number is using scientific notation and find dots
206		eIndex := -1
207		pIndex := -1
208		for i, r := range value {
209			if r == 'E' || r == 'e' {
210				if eIndex > -1 {
211					return Decimal{}, fmt.Errorf("can't convert %s to decimal: multiple 'E' characters found", value)
212				}
213				eIndex = i
214				continue
215			}
216	
217			if r == '.' {
218				if pIndex > -1 {
219					return Decimal{}, fmt.Errorf("can't convert %s to decimal: too many .s", value)
220				}
221				pIndex = i
222			}
223		}
224	
225		if eIndex != -1 {
226			expInt, err := strconv.ParseInt(value[eIndex+1:], 10, 32)
227			if err != nil {
228				if e, ok := err.(*strconv.NumError); ok && e.Err == strconv.ErrRange {
229					return Decimal{}, fmt.Errorf("can't convert %s to decimal: fractional part too long", value)
230				}
231				return Decimal{}, fmt.Errorf("can't convert %s to decimal: exponent is not numeric", value)
232			}
233			value = value[:eIndex]
234			exp = expInt
235		}
236	
237		if pIndex == -1 {
238			// There is no decimal point, we can just parse the original string as
239			// an int
240			intString = value
241		} else {
242			if pIndex+1 < len(value) {
243				intString = value[:pIndex] + value[pIndex+1:]
244			} else {
245				intString = value[:pIndex]
246			}
247			expInt := -len(value[pIndex+1:])
248			exp += int64(expInt)
249		}
250	
251		var dValue *big.Int
252		// strconv.ParseInt is faster than new(big.Int).SetString so this is just a shortcut for strings we know won't overflow
253		if len(intString) <= 18 {
254			parsed64, err := strconv.ParseInt(intString, 10, 64)
255			if err != nil {
256				return Decimal{}, fmt.Errorf("can't convert %s to decimal", value)
257			}
258			dValue = big.NewInt(parsed64)
259		} else {
260			dValue = new(big.Int)
261			_, ok := dValue.SetString(intString, 10)
262			if !ok {
263				return Decimal{}, fmt.Errorf("can't convert %s to decimal", value)
264			}
265		}
266	
267		if exp < math.MinInt32 || exp > math.MaxInt32 {
268			// NOTE(vadim): I doubt a string could realistically be this long
269			return Decimal{}, fmt.Errorf("can't convert %s to decimal: fractional part too long", originalInput)
270		}
271	
272		return Decimal{
273			value: dValue,
274			exp:   int32(exp),
275		}, nil
276	}

352	func newFromFloat(val float64, bits uint64, flt *floatInfo) Decimal {
353		if math.IsNaN(val) || math.IsInf(val, 0) {
354			panic(fmt.Sprintf("Cannot create a Decimal from %v", val))
355		}
356		exp := int(bits>>flt.mantbits) & (1<<flt.expbits - 1)
357		mant := bits & (uint64(1)<<flt.mantbits - 1)
358	
359		switch exp {
360		case 0:
361			// denormalized
362			exp++
363	
364		default:
365			// add implicit top bit
366			mant |= uint64(1) << flt.mantbits
367		}
368		exp += flt.bias
369	
370		var d decimal
371		d.Assign(mant)
372		d.Shift(exp - int(flt.mantbits))
373		d.neg = bits>>(flt.expbits+flt.mantbits) != 0
374	
375		roundShortest(&d, mant, exp, flt)
376		// If less than 19 digits, we can do calculation in an int64.
377		if d.nd < 19 {
378			tmp := int64(0)
379			m := int64(1)
380			for i := d.nd - 1; i >= 0; i-- {
381				tmp += m * int64(d.d[i]-'0')
382				m *= 10
383			}
384			if d.neg {
385				tmp *= -1
386			}
387			return Decimal{value: big.NewInt(tmp), exp: int32(d.dp) - int32(d.nd)}
388		}
389		dValue := new(big.Int)
390		dValue, ok := dValue.SetString(string(d.d[:d.nd]), 10)
391		if ok {
392			return Decimal{value: dValue, exp: int32(d.dp) - int32(d.nd)}
393		}
394	
395		return NewFromFloatWithExponent(val, int32(d.dp)-int32(d.nd))
396	}

398	// NewFromFloatWithExponent converts a float64 to Decimal, with an arbitrary
399	// number of fractional digits.
400	//
401	// Example:
402	//
403	//	NewFromFloatWithExponent(123.456, -2).String() // output: "123.46"
404	func NewFromFloatWithExponent(value float64, exp int32) Decimal {
405		if math.IsNaN(value) || math.IsInf(value, 0) {
406			panic(fmt.Sprintf("Cannot create a Decimal from %v", value))
407		}
408	
409		bits := math.Float64bits(value)
410		mant := bits & (1<<52 - 1)
411		exp2 := int32((bits >> 52) & (1<<11 - 1))
412		sign := bits >> 63
413	
414		if exp2 == 0 {
415			// specials
416			if mant == 0 {
417				return Decimal{}
418			}
419			// subnormal
420			exp2++
421		} else {
422			// normal
423			mant |= 1 << 52
424		}
425	
426		exp2 -= 1023 + 52
427	
428		// normalizing base-2 values
429		for mant&1 == 0 {
430			mant = mant >> 1
431			exp2++
432		}
433	
434		// maximum number of fractional base-10 digits to represent 2^N exactly cannot be more than -N if N<0
435		if exp < 0 && exp < exp2 {
436			if exp2 < 0 {
437				exp = exp2
438			} else {
439				exp = 0
440			}
441		}
442	
443		// representing 10^M * 2^N as 5^M * 2^(M+N)
444		exp2 -= exp
445	
446		temp := big.NewInt(1)
447		dMant := big.NewInt(int64(mant))
448	
449		// applying 5^M
450		if exp > 0 {
451			temp = temp.SetInt64(int64(exp))
452			temp = temp.Exp(fiveInt, temp, nil)
453		} else if exp < 0 {
454			temp = temp.SetInt64(-int64(exp))
455			temp = temp.Exp(fiveInt, temp, nil)
456			dMant = dMant.Mul(dMant, temp)
457			temp = temp.SetUint64(1)
458		}
459	
460		// applying 2^(M+N)
461		if exp2 > 0 {
462			dMant = dMant.Lsh(dMant, uint(exp2))
463		} else if exp2 < 0 {
464			temp = temp.Lsh(temp, uint(-exp2))
465		}
466	
467		// rounding and downscaling
468		if exp > 0 || exp2 < 0 {
469			halfDown := new(big.Int).Rsh(temp, 1)
470			dMant = dMant.Add(dMant, halfDown)
471			dMant = dMant.Quo(dMant, temp)
472		}
473	
474		if sign == 1 {
475			dMant = dMant.Neg(dMant)
476		}
477	
478		return Decimal{
479			value: dMant,
480			exp:   exp,
481		}
482	}

493	// decimal may be less precise if the given exponent is bigger
494	// than the initial exponent of the Decimal.
495	// NOTE: this will truncate, NOT round
496	//
497	// Example:
498	//
499	//	d := New(12345, -4)
500	//	d2 := d.rescale(-1)
501	//	d3 := d2.rescale(-4)
502	//	println(d1)
503	//	println(d2)
504	//	println(d3)
505	//
506	// Output:
507	//
508	//	1.2345

510	//	1.2000
511	func (d Decimal) rescale(exp int32) Decimal {
512		if d.exp == exp {
513			return Decimal{
514				new(big.Int).Set(d.getValue()),
515				d.exp,
516			}
517		}
518	
519		// NOTE(vadim): must convert exps to float64 before - to prevent overflow
520		diff := math.Abs(float64(exp) - float64(d.exp))
521		value := new(big.Int).Set(d.getValue())
522	
523		expScale := new(big.Int).Exp(tenInt, big.NewInt(int64(diff)), nil)
524		if exp > d.exp {
525			value = value.Quo(value, expScale)
526		} else if exp < d.exp {
527			value = value.Mul(value, expScale)
528		}
529	
530		return Decimal{
531			value: value,
532			exp:   exp,
533		}
534	}

612	// QuoRem does division with remainder
613	// d.QuoRem(d2,precision) returns quotient q and remainder r such that
614	//
615	//	d = d2 * q + r, q an integer multiple of 10^(-precision)
616	//	0 <= r < abs(d2) * 10 ^(-precision) if d>=0
617	//	0 >= r > -abs(d2) * 10 ^(-precision) if d<0
618	//
619	// Note that precision<0 is allowed as input.
620	func (d Decimal) QuoRem(d2 Decimal, precision int32) (Decimal, Decimal) {
621		if d2.getValue().Sign() == 0 {
622			panic("decimal division by 0")
623		}
624		scale := -precision
625		e := int64(d.exp) - int64(d2.exp) - int64(scale)
626		if e > math.MaxInt32 || e < math.MinInt32 {
627			panic("overflow in decimal QuoRem")
628		}
629		var aa, bb, expo big.Int
630		var scalerest int32
631		// d = a 10^ea
632		// d2 = b 10^eb
633		if e < 0 {
634			aa = *d.getValue()
635			expo.SetInt64(-e)
636			bb.Exp(tenInt, &expo, nil)
637			bb.Mul(d2.getValue(), &bb)
638			scalerest = d.exp
639			// now aa = a
640			//     bb = b 10^(scale + eb - ea)
641		} else {
642			expo.SetInt64(e)
643			aa.Exp(tenInt, &expo, nil)
644			aa.Mul(d.getValue(), &aa)
645			bb = *d2.getValue()
646			scalerest = scale + d2.exp
647			// now aa = a ^ (ea - eb - scale)
648			//     bb = b
649		}
650		var q, r big.Int
651		q.QuoRem(&aa, &bb, &r)
652		dq := Decimal{value: &q, exp: scale}
653		dr := Decimal{value: &r, exp: scalerest}
654		return dq, dr
655	}

657	// DivRound divides and rounds to a given precision
658	// i.e. to an integer multiple of 10^(-precision)
659	//
660	//	for a positive quotient digit 5 is rounded up, away from 0
661	//	if the quotient is negative then digit 5 is rounded down, away from 0
662	//
663	// Note that precision<0 is allowed as input.
664	func (d Decimal) DivRound(d2 Decimal, precision int32) Decimal {
665		// QuoRem already checks initialization
666		q, r := d.QuoRem(d2, precision)
667		// the actual rounding decision is based on comparing r*10^precision and d2/2
668		// instead compare 2 r 10 ^precision and d2
669		var rv2 big.Int
670		rv2.Abs(r.getValue())
671		rv2.Lsh(&rv2, 1)
672		// now rv2 = abs(r.value) * 2
673		r2 := Decimal{value: &rv2, exp: r.exp + precision}
674		// r2 is now 2 * r * 10 ^ precision
675		var c = r2.Cmp(d2.Abs())
676	
677		if c < 0 {
678			return q
679		}
680	
681		if d.getValue().Sign()*d2.getValue().Sign() < 0 {
682			return q.Sub(New(1, -precision))
683		}
684	
685		return q.Add(New(1, -precision))
686	}

691		return r

694	// Pow returns d to the power of d2.
695	// When exponent is negative the returned decimal will have maximum precision of PowPrecisionNegativeExponent places after decimal point.
696	//
697	// Pow returns 0 (zero-value of Decimal) instead of error for power operation edge cases, to handle those edge cases use PowWithPrecision
698	// Edge cases not handled by Pow:
699	//   - 0 ** 0 => undefined value
700	//   - 0 ** y, where y < 0 => infinity
701	//   - x ** y, where x < 0 and y is non-integer decimal => imaginary value
702	//
703	// Example:
704	//
705	//	d1 := decimal.NewFromFloat(4.0)
706	//	d2 := decimal.NewFromFloat(4.0)
707	//	res1 := d1.Pow(d2)
708	//	res1.String() // output: "256"
709	//
710	//	d3 := decimal.NewFromFloat(5.0)
711	//	d4 := decimal.NewFromFloat(5.73)
712	//	res2 := d3.Pow(d4)
713	//	res2.String() // output: "10118.08037125"
714	func (d Decimal) Pow(d2 Decimal) Decimal {
715		baseSign := d.Sign()
716		expSign := d2.Sign()
717	
718		if baseSign == 0 {
719			if expSign == 0 {
720				return Decimal{}
721			}
722			if expSign == 1 {
723				return Decimal{zeroInt, 0}
724			}
725			if expSign == -1 {
726				return Decimal{}
727			}
728		}
729	
730		if expSign == 0 {
731			return Decimal{oneInt, 0}
732		}
733	
734		// TODO: optimize extraction of fractional part
735		one := Decimal{oneInt, 0}
736		expIntPart, expFracPart := d2.QuoRem(one, 0)
737	
738		if baseSign == -1 && !expFracPart.IsZero() {
739			return Decimal{}
740		}
741	
742		intPartPow, _ := d.PowBigInt(expIntPart.getValue())
743	
744		// if exponent is an integer we don't need to calculate d1**frac(d2)
745		if expFracPart.getValue().Sign() == 0 {
746			return intPartPow
747		}
748	
749		// TODO: optimize NumDigits for more performant precision adjustment
750		digitsBase := d.NumDigits()
751		digitsExponent := d2.NumDigits()
752	
753		precision := digitsBase
754	
755		if digitsExponent > precision {
756			precision += digitsExponent
757		}
758	
759		precision += 6
760	
761		// Calculate x ** frac(y), where
762		// x ** frac(y) = exp(ln(x ** frac(y)) = exp(ln(x) * frac(y))
763		fracPartPow, err := d.Abs().Ln(-d.exp + int32(precision))
764		if err != nil {
765			return Decimal{}
766		}
767	
768		fracPartPow = fracPartPow.Mul(expFracPart)
769	
770		fracPartPow, err = fracPartPow.ExpTaylor(-d.exp + int32(precision))
771		if err != nil {
772			return Decimal{}
773		}
774	
775		// Join integer and fractional part,
776		// base ** (expBase + expFrac) = base ** expBase * base ** expFrac
777		res := intPartPow.Mul(fracPartPow)
778	
779		return res
780	}

782	// PowWithPrecision returns d to the power of d2.
783	// Precision parameter specifies minimum precision of the result (digits after decimal point).
784	// Returned decimal is not rounded to 'precision' places after decimal point.
785	//
786	// PowWithPrecision returns error when:
787	//   - 0 ** 0 => undefined value
788	//   - 0 ** y, where y < 0 => infinity
789	//   - x ** y, where x < 0 and y is non-integer decimal => imaginary value
790	//
791	// Example:
792	//
793	//	d1 := decimal.NewFromFloat(4.0)
794	//	d2 := decimal.NewFromFloat(4.0)
795	//	res1, err := d1.PowWithPrecision(d2, 2)
796	//	res1.String() // output: "256"
797	//
798	//	d3 := decimal.NewFromFloat(5.0)
799	//	d4 := decimal.NewFromFloat(5.73)
800	//	res2, err := d3.PowWithPrecision(d4, 5)
801	//	res2.String() // output: "10118.080371595015625"
802	//
803	//	d5 := decimal.NewFromFloat(-3.0)
804	//	d6 := decimal.NewFromFloat(-6.0)
805	//	res3, err := d5.PowWithPrecision(d6, 10)
806	//	res3.String() // output: "0.0013717421"
807	func (d Decimal) PowWithPrecision(d2 Decimal, precision int32) (Decimal, error) {
808		baseSign := d.Sign()
809		expSign := d2.Sign()
810	
811		if baseSign == 0 {
812			if expSign == 0 {
813				return Decimal{}, fmt.Errorf("cannot represent undefined value of 0**0")
814			}
815			if expSign == 1 {
816				return Decimal{zeroInt, 0}, nil
817			}
818			if expSign == -1 {
819				return Decimal{}, fmt.Errorf("cannot represent infinity value of 0 ** y, where y < 0")
820			}
821		}
822	
823		if expSign == 0 {
824			return Decimal{oneInt, 0}, nil
825		}
826	
827		// TODO: optimize extraction of fractional part
828		one := Decimal{oneInt, 0}
829		expIntPart, expFracPart := d2.QuoRem(one, 0)
830	
831		if baseSign == -1 && !expFracPart.IsZero() {
832			return Decimal{}, fmt.Errorf("cannot represent imaginary value of x ** y, where x < 0 and y is non-integer decimal")
833		}
834	
835		intPartPow, _ := d.powBigIntWithPrecision(expIntPart.getValue(), precision)
836	
837		// if exponent is an integer we don't need to calculate d1**frac(d2)
838		if expFracPart.getValue().Sign() == 0 {
839			return intPartPow, nil
840		}
841	
842		// TODO: optimize NumDigits for more performant precision adjustment
843		digitsBase := d.NumDigits()
844		digitsExponent := d2.NumDigits()
845	
846		if int32(digitsBase) > precision {
847			precision = int32(digitsBase)
848		}
849		if int32(digitsExponent) > precision {
850			precision += int32(digitsExponent)
851		}
852		// increase precision by 10 to compensate for errors in further calculations
853		precision += 10
854	
855		// Calculate x ** frac(y), where
856		// x ** frac(y) = exp(ln(x ** frac(y)) = exp(ln(x) * frac(y))
857		fracPartPow, err := d.Abs().Ln(precision)
858		if err != nil {
859			return Decimal{}, err
860		}
861	
862		fracPartPow = fracPartPow.Mul(expFracPart)
863	
864		fracPartPow, err = fracPartPow.ExpTaylor(precision)
865		if err != nil {
866			return Decimal{}, err
867		}
868	
869		// Join integer and fractional part,
870		// base ** (expBase + expFrac) = base ** expBase * base ** expFrac
871		res := intPartPow.Mul(fracPartPow)
872	
873		return res, nil
874	}

876	// PowInt32 returns d to the power of exp, where exp is int32.
877	// Only returns error when d and exp is 0, thus result is undefined.
878	//
879	// When exponent is negative the returned decimal will have maximum precision of PowPrecisionNegativeExponent places after decimal point.
880	//
881	// Example:
882	//
883	//	d1, err := decimal.NewFromFloat(4.0).PowInt32(4)
884	//	d1.String() // output: "256"
885	//
886	//	d2, err := decimal.NewFromFloat(3.13).PowInt32(5)
887	//	d2.String() // output: "300.4150512793"
888	func (d Decimal) PowInt32(exp int32) (Decimal, error) {
889		if d.IsZero() && exp == 0 {
890			return Decimal{}, fmt.Errorf("cannot represent undefined value of 0**0")
891		}
892	
893		isExpNeg := exp < 0
894		exp = abs(exp)
895	
896		n, result := d, New(1, 0)
897	
898		for exp > 0 {
899			if exp%2 == 1 {
900				result = result.Mul(n)
901			}
902			exp /= 2
903	
904			if exp > 0 {
905				n = n.Mul(n)
906			}
907		}
908	
909		if isExpNeg {
910			return New(1, 0).DivRound(result, int32(PowPrecisionNegativeExponent)), nil
911		}
912	
913		return result, nil
914	}

932	func (d Decimal) powBigIntWithPrecision(exp *big.Int, precision int32) (Decimal, error) {
933		if d.IsZero() && exp.Sign() == 0 {
934			return Decimal{}, fmt.Errorf("cannot represent undefined value of 0**0")
935		}
936	
937		tmpExp := new(big.Int).Set(exp)
938		isExpNeg := exp.Sign() < 0
939	
940		if isExpNeg {
941			tmpExp.Abs(tmpExp)
942		}
943	
944		n, result := d, New(1, 0)
945	
946		for tmpExp.Sign() > 0 {
947			if tmpExp.Bit(0) == 1 {
948				result = result.Mul(n)
949			}
950			tmpExp.Rsh(tmpExp, 1)
951	
952			if tmpExp.Sign() > 0 {
953				n = n.Mul(n)
954			}
955		}
956	
957		if isExpNeg {
958			return New(1, 0).DivRound(result, precision), nil
959		}
960	
961		return result, nil
962	}

964	// ExpHullAbrham calculates the natural exponent of decimal (e to the power of d) using Hull-Abraham algorithm.
965	// OverallPrecision argument specifies the overall precision of the result (integer part + decimal part).
966	//
967	// ExpHullAbrham is faster than ExpTaylor for small precision values, but it is much slower for large precision values.
968	//
969	// Example:
970	//
971	//	NewFromFloat(26.1).ExpHullAbrham(2).String()    // output: "220000000000"
972	//	NewFromFloat(26.1).ExpHullAbrham(20).String()   // output: "216314672147.05767284"
973	func (d Decimal) ExpHullAbrham(overallPrecision uint32) (Decimal, error) {
974		// Algorithm based on Variable precision exponential function.
975		// ACM Transactions on Mathematical Software by T. E. Hull & A. Abrham.
976		if d.IsZero() {
977			return Decimal{oneInt, 0}, nil
978		}
979	
980		currentPrecision := overallPrecision
981	
982		// Algorithm does not work if currentPrecision * 23 < |x|.
983		// Precision is automatically increased in such cases, so the value can be calculated precisely.
984		// If newly calculated precision is higher than ExpMaxIterations the currentPrecision will not be changed.
985		f := d.Abs().InexactFloat64()
986		if ncp := f / 23; ncp > float64(currentPrecision) && ncp < float64(ExpMaxIterations) {
987			currentPrecision = uint32(math.Ceil(ncp))
988		}
989	
990		// fail if abs(d) beyond an over/underflow threshold
991		overflowThreshold := New(23*int64(currentPrecision), 0)
992		if d.Abs().Cmp(overflowThreshold) > 0 {
993			return Decimal{}, fmt.Errorf("over/underflow threshold, exp(x) cannot be calculated precisely")
994		}
995	
996		// Return 1 if abs(d) small enough; this also avoids later over/underflow
997		overflowThreshold2 := New(9, -int32(currentPrecision)-1)
998		if d.Abs().Cmp(overflowThreshold2) <= 0 {
999			return Decimal{oneInt, d.exp}, nil
1000		}
1001	
1002		// t is the smallest integer >= 0 such that the corresponding abs(d/k) < 1
1003		t := d.exp + int32(d.NumDigits()) // Add d.NumDigits because the paper assumes that d.value [0.1, 1)
1004	
1005		if t < 0 {
1006			t = 0
1007		}
1008	
1009		k := New(1, t)                                          // reduction factor
1010		r := Decimal{new(big.Int).Set(d.getValue()), d.exp - t} // reduced argument
1011		p := int32(currentPrecision) + t + 2                    // precision for calculating the sum
1012	
1013		// Determine n, the number of therms for calculating sum
1014		// use first Newton step (1.435p - 1.182) / log10(p/abs(r))
1015		// for solving appropriate equation, along with directed
1016		// roundings and simple rational bound for log10(p/abs(r))
1017		rf := r.Abs().InexactFloat64()
1018		pf := float64(p)
1019		nf := math.Ceil((1.453*pf - 1.182) / math.Log10(pf/rf))
1020		if nf > float64(ExpMaxIterations) || math.IsNaN(nf) {
1021			return Decimal{}, fmt.Errorf("exact value cannot be calculated in <=ExpMaxIterations iterations")
1022		}
1023		n := int64(nf)
1024	
1025		tmp := New(0, 0)
1026		sum := New(1, 0)
1027		one := New(1, 0)
1028		for i := n - 1; i > 0; i-- {
1029			tmp.value.SetInt64(i)
1030			sum = sum.Mul(r.DivRound(tmp, p))
1031			sum = sum.Add(one)
1032		}
1033	
1034		ki := k.IntPart()
1035		res := New(1, 0)
1036		for i := ki; i > 0; i-- {
1037			res = res.Mul(sum)
1038		}
1039	
1040		resNumDigits := int32(res.NumDigits())
1041	
1042		var roundDigits int32
1043		if resNumDigits > abs(res.exp) {
1044			roundDigits = int32(currentPrecision) - resNumDigits - res.exp
1045		} else {
1046			roundDigits = int32(currentPrecision)
1047		}
1048	
1049		res = res.Round(roundDigits)
1050	
1051		return res, nil
1052	}

1054	// ExpTaylor calculates the natural exponent of decimal (e to the power of d) using Taylor series expansion.
1055	// Precision argument specifies how precise the result must be (number of digits after decimal point).
1056	// Negative precision is allowed.
1057	//
1058	// ExpTaylor is much faster for large precision values than ExpHullAbrham.
1059	//
1060	// Example:
1061	//
1062	//	d, err := NewFromFloat(26.1).ExpTaylor(2).String()
1063	//	d.String()  // output: "216314672147.06"
1064	//
1065	//	NewFromFloat(26.1).ExpTaylor(20).String()
1066	//	d.String()  // output: "216314672147.05767284062928674083"
1067	//
1068	//	NewFromFloat(26.1).ExpTaylor(-10).String()
1069	//	d.String()  // output: "220000000000"
1070	func (d Decimal) ExpTaylor(precision int32) (Decimal, error) {
1071		// Note(mwoss): Implementation can be optimized by exclusively using big.Int API only
1072		if d.IsZero() {
1073			return Decimal{oneInt, 0}.Round(precision), nil
1074		}
1075	
1076		var epsilon Decimal
1077		var divPrecision int32
1078		if precision < 0 {
1079			epsilon = New(1, -1)
1080			divPrecision = 8
1081		} else {
1082			epsilon = New(1, -precision-1)
1083			divPrecision = precision + 1
1084		}
1085	
1086		decAbs := d.Abs()
1087		pow := d.Abs()
1088		factorial := New(1, 0)
1089	
1090		result := New(1, 0)
1091	
1092		for i := int64(1); ; {
1093			step := pow.DivRound(factorial, divPrecision)
1094			result = result.Add(step)
1095	
1096			// Stop Taylor series when current step is smaller than epsilon
1097			if step.Cmp(epsilon) < 0 {
1098				break
1099			}
1100	
1101			pow = pow.Mul(decAbs)
1102	
1103			i++
1104	
1105			// Calculate next factorial number or retrieve cached value
1106			if len(factorials) >= int(i) && !factorials[i-1].IsZero() {
1107				factorial = factorials[i-1]
1108			} else {
1109				// To avoid any race conditions, firstly the zero value is appended to a slice to create
1110				// a spot for newly calculated factorial. After that, the zero value is replaced by calculated
1111				// factorial using the index notation.
1112				factorial = factorials[i-2].Mul(New(i, 0))
1113				factorials = append(factorials, Zero)
1114				factorials[i-1] = factorial
1115			}
1116		}
1117	
1118		if d.Sign() < 0 {
1119			result = New(1, 0).DivRound(result, precision+1)
1120		}
1121	
1122		result = result.Round(precision)
1123		return result, nil
1124	}
1125	
1126	// Ln calculates natural logarithm of d.
1127	// Precision argument specifies how precise the result must be (number of digits after decimal point).
1128	// Negative precision is allowed.
1129	//
1130	// Example:
1131	//
1132	//	d1, err := NewFromFloat(13.3).Ln(2)
1133	//	d1.String()  // output: "2.59"
1134	//
1135	//	d2, err := NewFromFloat(579.161).Ln(10)
1136	//	d2.String()  // output: "6.3615805046"
1137	func (d Decimal) Ln(precision int32) (Decimal, error) {
1138		// Algorithm based on The Use of Iteration Methods for Approximating the Natural Logarithm,
1139		// James F. Epperson, The American Mathematical Monthly, Vol. 96, No. 9, November 1989, pp. 831-835.
1140		if d.IsNegative() {
1141			return Decimal{}, fmt.Errorf("cannot calculate natural logarithm for negative decimals")
1142		}
1143	
1144		if d.IsZero() {
1145			return Decimal{}, fmt.Errorf("cannot represent natural logarithm of 0, result: -infinity")
1146		}
1147	
1148		calcPrecision := precision + 2
1149		z := d.Copy()
1150	
1151		var comp1, comp3, comp2, comp4, reduceAdjust Decimal
1152		comp1 = z.Sub(Decimal{oneInt, 0})
1153		comp3 = Decimal{oneInt, -1}
1154	
1155		// for decimal in range [0.9, 1.1] where ln(d) is close to 0
1156		usePowerSeries := false
1157	
1158		if comp1.Abs().Cmp(comp3) <= 0 {
1159			usePowerSeries = true
1160		} else {
1161			// reduce input decimal to range [0.1, 1)
1162			expDelta := int32(z.NumDigits()) + z.exp
1163			z.exp -= expDelta
1164	
1165			// Input decimal was reduced by factor of 10^expDelta, thus we will need to add
1166			// ln(10^expDelta) = expDelta * ln(10)
1167			// to the result to compensate that
1168			ln10 := ln10.withPrecision(calcPrecision)
1169			reduceAdjust = NewFromInt32(expDelta)
1170			reduceAdjust = reduceAdjust.Mul(ln10)
1171	
1172			comp1 = z.Sub(Decimal{oneInt, 0})
1173	
1174			if comp1.Abs().Cmp(comp3) <= 0 {
1175				usePowerSeries = true
1176			} else {
1177				// initial estimate using floats
1178				zFloat := z.InexactFloat64()
1179				comp1 = NewFromFloat(math.Log(zFloat))
1180			}
1181		}
1182	
1183		epsilon := Decimal{oneInt, -calcPrecision}
1184	
1185		if usePowerSeries {
1186			// Power Series - https://en.wikipedia.org/wiki/Logarithm#Power_series
1187			// Calculating n-th term of formula: ln(z+1) = 2 sum [ 1 / (2n+1) * (z / (z+2))^(2n+1) ]
1188			// until the difference between current and next term is smaller than epsilon.
1189			// Coverage quite fast for decimals close to 1.0
1190	
1191			// z + 2
1192			comp2 = comp1.Add(Decimal{twoInt, 0})
1193			// z / (z + 2)
1194			comp3 = comp1.DivRound(comp2, calcPrecision)
1195			// 2 * (z / (z + 2))
1196			comp1 = comp3.Add(comp3)
1197			comp2 = comp1.Copy()
1198	
1199			for n := 1; ; n++ {
1200				// 2 * (z / (z+2))^(2n+1)
1201				comp2 = comp2.Mul(comp3).Mul(comp3)
1202	
1203				// 1 / (2n+1) * 2 * (z / (z+2))^(2n+1)
1204				comp4 = NewFromInt(int64(2*n + 1))
1205				comp4 = comp2.DivRound(comp4, calcPrecision)
1206	
1207				// comp1 = 2 sum [ 1 / (2n+1) * (z / (z+2))^(2n+1) ]
1208				comp1 = comp1.Add(comp4)
1209	
1210				if comp4.Abs().Cmp(epsilon) <= 0 {
1211					break
1212				}
1213			}
1214		} else {
1215			// Halley's Iteration.
1216			// Calculating n-th term of formula: a_(n+1) = a_n - 2 * (exp(a_n) - z) / (exp(a_n) + z),
1217			// until the difference between current and next term is smaller than epsilon
1218			var prevStep Decimal
1219			maxIters := calcPrecision*2 + 10
1220	
1221			for i := int32(0); i < maxIters; i++ {
1222				// exp(a_n)
1223				comp3, _ = comp1.ExpTaylor(calcPrecision)
1224				// exp(a_n) - z
1225				comp2 = comp3.Sub(z)
1226				// 2 * (exp(a_n) - z)
1227				comp2 = comp2.Add(comp2)
1228				// exp(a_n) + z
1229				comp4 = comp3.Add(z)
1230				// 2 * (exp(a_n) - z) / (exp(a_n) + z)
1231				comp3 = comp2.DivRound(comp4, calcPrecision)
1232				// comp1 = a_(n+1) = a_n - 2 * (exp(a_n) - z) / (exp(a_n) + z)
1233				comp1 = comp1.Sub(comp3)
1234	
1235				if prevStep.Add(comp3).IsZero() {
1236					// If iteration steps oscillate we should return early and prevent an infinity loop
1237					// NOTE(mwoss): This should be quite a rare case, returning error is not necessary
1238					break
1239				}
1240	
1241				if comp3.Abs().Cmp(epsilon) <= 0 {
1242					break
1243				}
1244	
1245				prevStep = comp3
1246			}
1247		}
1248	
1249		comp1 = comp1.Add(reduceAdjust)
1250	
1251		return comp1.Round(precision), nil
1252	}

1254	// NumDigits returns the number of digits of the decimal coefficient (d.Value)
1255	func (d Decimal) NumDigits() int {
1256		v := d.getValue()
1257		if v.IsInt64() {
1258			i64 := v.Int64()
1259			// restrict fast path to integers with exact conversion to float64
1260			if i64 <= (1<<53) && i64 >= -(1<<53) {
1261				if i64 == 0 {
1262					return 1
1263				}
1264				return int(math.Log10(math.Abs(float64(i64)))) + 1
1265			}
1266		}
1267	
1268		estimatedNumDigits := int(float64(v.BitLen()) / math.Log2(10))
1269	
1270		// estimatedNumDigits (lg10) may be off by 1, need to verify
1271		digitsBigInt := big.NewInt(int64(estimatedNumDigits))
1272		errorCorrectionUnit := digitsBigInt.Exp(tenInt, digitsBigInt, nil)
1273	
1274		if v.CmpAbs(errorCorrectionUnit) >= 0 {
1275			return estimatedNumDigits + 1
1276		}
1277	
1278		return estimatedNumDigits
1279	}

1281	// IsInteger returns true when decimal can be represented as an integer value, otherwise, it returns false.
1282	func (d Decimal) IsInteger() bool {
1283		// The most typical case, all decimal with exponent higher or equal 0 can be represented as integer
1284		if d.exp >= 0 {
1285			return true
1286		}
1287		// When the exponent is negative we have to check every number after the decimal place
1288		// If all of them are zeroes, we are sure that given decimal can be represented as an integer
1289		var r big.Int
1290		q := new(big.Int).Set(d.getValue())
1291		for z := abs(d.exp); z > 0; z-- {
1292			q.QuoRem(q, tenInt, &r)
1293			if r.Cmp(zeroInt) != 0 {
1294				return false
1295			}
1296		}
1297		return true
1298	}

1526	// Round rounds the decimal to places decimal places.
1527	// If places < 0, it will round the integer part to the nearest 10^(-places).
1528	//
1529	// Example:
1530	//
1531	//	NewFromFloat(5.45).Round(1).String() // output: "5.5"
1532	//	NewFromFloat(545).Round(-1).String() // output: "550" (with UseScientificNotation false, "5.5E2" if true)
1533	func (d Decimal) Round(places int32) Decimal {
1534		if d.exp == -places {
1535			return d
1536		}
1537		// truncate to places + 1
1538		ret := d.rescale(-places - 1)
1539	
1540		// add sign(d) * 0.5
1541		if ret.value.Sign() < 0 {
1542			ret.value.Sub(ret.value, fiveInt)
1543		} else {
1544			ret.value.Add(ret.value, fiveInt)
1545		}
1546	
1547		// floor for positive numbers, ceil for negative numbers
1548		_, m := ret.value.DivMod(ret.value, tenInt, new(big.Int))
1549		ret.exp++
1550		if ret.value.Sign() < 0 && m.Cmp(zeroInt) != 0 {
1551			ret.value.Add(ret.value, oneInt)
1552		}
1553	
1554		return ret
1555	}

1654	// RoundBank rounds the decimal to places decimal places.
1655	// If the final digit to round is equidistant from the nearest two integers the
1656	// rounded value is taken as the even number
1657	//
1658	// If places < 0, it will round the integer part to the nearest 10^(-places).
1659	//
1660	// Examples:
1661	//
1662	//	NewFromFloat(5.45).RoundBank(1).String() // output: "5.4"
1663	//	NewFromFloat(545).RoundBank(-1).String() // output: "540"
1664	//	NewFromFloat(5.46).RoundBank(1).String() // output: "5.5"
1665	//	NewFromFloat(546).RoundBank(-1).String() // output: "550"
1666	//	NewFromFloat(5.55).RoundBank(1).String() // output: "5.6"
1667	//	NewFromFloat(555).RoundBank(-1).String() // output: "560"
1668	func (d Decimal) RoundBank(places int32) Decimal {
1669	
1670		round := d.Round(places)
1671		remainder := d.Sub(round).Abs()
1672	
1673		half := New(5, -places-1)
1674		if remainder.Cmp(half) == 0 && round.getValue().Bit(0) != 0 {
1675			if round.getValue().Sign() < 0 {
1676				round.value = new(big.Int).Add(round.getValue(), oneInt)
1677			} else {
1678				round.value = new(big.Int).Sub(round.getValue(), oneInt)
1679			}
1680		}
1681	
1682		return round
1683	}

1685	// RoundCash aka Cash/Penny/öre rounding rounds decimal to a specific
1686	// interval. The amount payable for a cash transaction is rounded to the nearest
1687	// multiple of the minimum currency unit available. The following intervals are
1688	// available: 5, 10, 25, 50 and 100; any other number throws a panic.
1689	//
1690	//	  5:   5 cent rounding 3.43 => 3.45
1691	//	 10:  10 cent rounding 3.45 => 3.50 (5 gets rounded up)
1692	//	 25:  25 cent rounding 3.41 => 3.50
1693	//	 50:  50 cent rounding 3.75 => 4.00
1694	//	100: 100 cent rounding 3.50 => 4.00
1695	//
1696	// For more details: https://en.wikipedia.org/wiki/Cash_rounding
1697	func (d Decimal) RoundCash(interval uint8) Decimal {
1698		var iVal *big.Int
1699		switch interval {
1700		case 5:
1701			iVal = twentyInt
1702		case 10:
1703			iVal = tenInt
1704		case 25:
1705			iVal = fourInt
1706		case 50:
1707			iVal = twoInt
1708		case 100:
1709			iVal = oneInt
1710		default:
1711			panic(fmt.Sprintf("Decimal does not support this Cash rounding interval `%d`. Supported: 5, 10, 25, 50, 100", interval))
1712		}
1713		dVal := Decimal{
1714			value: iVal,
1715		}
1716	
1717		// TODO: optimize those calculations to reduce the high allocations (~29 allocs).
1718		return d.Mul(dVal).Round(0).Div(dVal).Truncate(2)
1719	}

1798		if len(data) < 4 {

1807		if err := d.value.GobDecode(data[4:]); err != nil {
1808			return fmt.Errorf("error decoding binary %v: %s", data, err)
1809		}
1810	
1811		return nil
1812	}

1815	func (d Decimal) MarshalBinary() (data []byte, err error) {
1816		// exp is written first, but encode value first to know output size
1817		var valueData []byte
1818		if valueData, err = d.getValue().GobEncode(); err != nil {
1819			return nil, err

1826		// Return the byte array

1922	func (d Decimal) string(trimTrailingZeros, useScientificNotation bool) string {
1923		if d.exp == 0 {
1924			return d.rescale(0).getValue().String()
1925		}
1926		if d.exp >= 0 {
1927			if useScientificNotation {
1928				return d.ScientificNotationString()
1929			} else {
1930				return d.rescale(0).value.String()
1931			}
1932		}
1933	
1934		abs := new(big.Int).Abs(d.getValue())
1935		str := abs.String()
1936	
1937		var intPart, fractionalPart string
1938	
1939		// NOTE(vadim): this cast to int will cause bugs if d.exp == INT_MIN
1940		// and you are on a 32-bit machine. Won't fix this super-edge case.
1941		dExpInt := int(d.exp)
1942		if len(str) > -dExpInt {
1943			intPart = str[:len(str)+dExpInt]
1944			fractionalPart = str[len(str)+dExpInt:]
1945		} else {
1946			intPart = "0"
1947	
1948			num0s := -dExpInt - len(str)
1949			fractionalPart = strings.Repeat("0", num0s) + str
1950		}
1951	
1952		if trimTrailingZeros {
1953			i := len(fractionalPart) - 1
1954			for ; i >= 0; i-- {
1955				if fractionalPart[i] != '0' {
1956					break
1957				}
1958			}
1959			fractionalPart = fractionalPart[:i+1]
1960		}
1961	
1962		number := intPart
1963		if len(fractionalPart) > 0 {
1964			number += "." + fractionalPart
1965		}
1966	
1967		if d.getValue().Sign() < 0 {
1968			return "-" + number
1969		}
1970	
1971		return number
1972	}

1974	// ScientificNotationString serializes the decimal into standard scientific notation.
1975	//
1976	// The notation is normalized to have one non-zero digit followed by a decimal point and
1977	// the remaining significant digits followed by "E" and the base-10 exponent.
1978	//
1979	// A zero, which has no significant digits, is simply serialized to "0".
1980	func (d Decimal) ScientificNotationString() string {
1981		exp := int(d.exp)
1982		intStr := new(big.Int).Abs(d.getValue()).String()
1983		if intStr == "0" {
1984			return intStr
1985		}
1986		first := intStr[0]
1987		var remaining string
1988		if len(intStr) > 1 {
1989			remaining = "." + intStr[1:]
1990			exp = exp + len(intStr) - 1
1991		}
1992		number := string(first) + remaining + "E" + strconv.Itoa(exp)
1993		if d.value.Sign() < 0 {
1994			return "-" + number
1995		}
1996		return number
1997	}
