284	func (s *threadUnsafeSet[T]) PopN(n int) (items []T, count int) {
285		if n <= 0 || len(*s) == 0 {
286			return make([]T, 0), 0
287		}
288		sn := s.Cardinality()
289		if n > sn {
290			n = sn
291		}
292	
293		items = make([]T, 0, n)
294		for item := range *s {
295			if count >= n {
296				break
297			}
298			delete(*s, item)
299			items = append(items, item)
300			count++
301		}
302		return items, count
303	}

324	func (s *threadUnsafeSet[T]) SymmetricDifference(other Set[T]) Set[T] {
325		o := other.(*threadUnsafeSet[T])
326	
327		// maximum number of elements is the sum of s and o cardinalities (when s and o are disjoint)
328		n := s.Cardinality() + o.Cardinality()
329		sd := make(threadUnsafeSet[T], n)
330		for elem := range *s {
331			if !o.contains(elem) {
332				sd.add(elem)
333			}
334		}
335		for elem := range *o {
336			if !s.contains(elem) {
337				sd.add(elem)
338			}
339		}
340		return &sd
341	}
