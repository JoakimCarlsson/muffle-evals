109	    return f"(?:{s})"

112	def joinstem(cutpoint: int | None = 0, words: Iterable[str] | None = None) -> str:
113	    """
114	    Join stem of each word in words into a string for regex.
115	
116	    Each word is truncated at cutpoint.
117	
118	    Cutpoint is usually negative indicating the number of letters to remove
119	    from the end of each word.
120	
121	    >>> joinstem(-2, ["ephemeris", "iris", ".*itis"])
122	    '(?:ephemer|ir|.*it)'
123	
124	    >>> joinstem(None, ["ephemeris"])
125	    '(?:ephemeris)'
126	
127	    >>> joinstem(5, None)
128	    '(?:)'
129	    """
130	    return enclose("|".join(w[:cutpoint] for w in words or []))

133	def bysize(words: Iterable[str]) -> dict[int, set]:
134	    """
135	    From a list of words, return a dict of sets sorted by word length.
136	
137	    >>> words = ['ant', 'cat', 'dog', 'pig', 'frog', 'goat', 'horse', 'elephant']
138	    >>> ret = bysize(words)
139	    >>> sorted(ret[3])
140	    ['ant', 'cat', 'dog', 'pig']
141	    >>> ret[5]
142	    {'horse'}
143	    """
144	    res: dict[int, set] = collections.defaultdict(set)
145	    for w in words:
146	        res[len(w)].add(w)
147	    return res

150	def make_pl_si_lists(
151	    lst: Iterable[str],
152	    plending: str,
153	    siendingsize: int | None,
154	    dojoinstem: bool = True,
155	):
156	    """
157	    given a list of singular words: lst
158	
159	    an ending to append to make the plural: plending
160	
161	    the number of characters to remove from the singular
162	    before appending plending: siendingsize
163	
164	    a flag whether to create a joinstem: dojoinstem
165	
166	    return:
167	    a list of pluralised words: si_list (called si because this is what you need to
168	    look for to make the singular)
169	
170	    the pluralised words as a dict of sets sorted by word length: si_bysize
171	    the singular words as a dict of sets sorted by word length: pl_bysize
172	    if dojoinstem is True: a regular expression that matches any of the stems: stem
173	    """
174	    if siendingsize is not None:
175	        siendingsize = -siendingsize
176	    si_list = [w[:siendingsize] + plending for w in lst]
177	    pl_bysize = bysize(lst)
178	    si_bysize = bysize(si_list)
179	    if dojoinstem:
180	        stem = joinstem(siendingsize, lst)
181	        return si_list, si_bysize, pl_bysize, stem
182	    else:
183	        return si_list, si_bysize, pl_bysize

2022	def _grouped_from_right(digits: str, size: int = 3) -> Iterator[str]:
2023	    """
2024	    Yield ``digits`` in groups of ``size`` characters, aligned from the right.
2025	
2026	    >>> list(_grouped_from_right("1234567"))
2027	    ['1', '234', '567']
2028	    """
2029	    head = (len(digits) - 1) % size + 1
2030	    yield digits[:head]
2031	    yield from (digits[i : i + size] for i in range(head, len(digits), size))

2034	def _passthrough_sign(func: Callable[..., str]) -> Callable[..., str]:
2035	    """
2036	    Wrap a magnitude transform so a leading ``+``/``-`` sign passes through.
2037	
2038	    The sign is stripped from the first argument before the call and restored

2401	    def postprocess(self, orig: str, inflected) -> str:
2402	        inflected = str(inflected)
2403	        if "|" in inflected:
2404	            word_options = inflected.split("|")
2405	            # When two parts of a noun need to be pluralized
2406	            if len(word_options[0].split(" ")) == len(word_options[1].split(" ")):
2407	                result = inflected.split("|")[self.classical_dict["all"]].split(" ")
2408	            # When only the last part of the noun needs to be pluralized
2409	            else:
2410	                result = inflected.split(" ")
2411	                for index, word in enumerate(result):
2412	                    if "|" in word:
2413	                        result[index] = word.split("|")[self.classical_dict["all"]]
2414	        else:
2415	            result = inflected.split(" ")
2416	
2417	        # Try to fix word wise capitalization by restoring each inflected
2418	        # word's case from the original word it came from. Inflecting a phrase
2419	        # can change the number of words (e.g. "a cat" -> "cats" drops the
2420	        # article), so pair the words positionally only as far as they align
2421	        # and leave any surplus inflected words untouched rather than indexing
2422	        # past the end of either sequence.
2423	        originals = orig.split(" ")
2424	        recased = (
2425	            self._recase(original, word) for original, word in zip(originals, result)
2426	        )
2427	        surplus = result[len(originals) :]
2428	        return " ".join(itertools.chain(recased, surplus))

2430	    def _recase(self, original: str, inflected: str) -> str:
2431	        """
2432	        Return ``inflected`` recased to match the case of ``original``.
2433	        """
2434	        if original in self.pl_sb_preserve_case:
2435	            # This word's capitalization is intrinsic (e.g. "I", or an
2436	            # acronym registered via ``defnoun(..., case='preserve')``), so
2437	            # keep the inflected case rather than normalizing it to the case
2438	            # of the input.
2439	            return inflected
2440	        if original == original.upper():
2441	            return inflected.upper()
2442	        if original == original.capitalize():
2443	            return inflected.capitalize()
2444	        return inflected
