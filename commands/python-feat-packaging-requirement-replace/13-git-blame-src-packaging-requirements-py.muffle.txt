27	class InvalidRequirement(ValueError):
28	    """
29	    Raised when attempting to create a :class:`Requirement` with a string that
30	    does not conform to the specification. Users should refer to PEP 508.
31	
32	    .. versionadded:: 16.1
33	    """

36	class Requirement:
37	    """Represent a requirement for a project.
38	
39	    Parse a given requirement string into its parts, such as name, specifier,
40	    URL, and extras, according to the specification.
41	
42	    .. versionadded:: 16.1
43	
44	    .. versionchanged:: 22.0
45	        Added equality (``__eq__``) and hashing (``__hash__``) so requirements
46	        can be compared and stored in sets / dicts.
47	
48	    .. versionchanged:: 23.2
49	        Equality and hashing began canonicalizing requirement names, so
50	        requirements whose names differ only by normalization (e.g.
51	        ``Requirement("Foo")`` vs ``Requirement("foo")``) now compare and hash
52	        equal.
53	
54	    Instances are safe to serialize with :mod:`pickle`. They use a stable
55	    format so the same pickle can be loaded in future packaging releases.
56	
57	    .. versionchanged:: 26.2
58	
59	        Added a stable pickle format. Pickles created with packaging 26.2+ can
60	        be unpickled with future releases.  Backward compatibility with pickles
61	        from packaging < 26.2 is supported but may be removed in a future
62	        release.
63	
64	    .. versionchanged:: 26.3
65	
66	        The dedicated pickle support introduced in 26.2 did not preserve the
67	        specifier's explicit :attr:`~packaging.specifiers.SpecifierSet.prereleases`
68	        override; it is now included again.
69	
70	        Equality and hashing normalize requirement names, extras, and
71	        equivalent specifiers. The string representation still preserves the
72	        parsed name and extras spelling.
73	
74	    :param str requirement_string: The string representation of a requirement.
75	    :raises InvalidRequirement: If the given ``requirement_string`` is not
76	        parseable, then this exception will be raised.
77	    """
78	
79	    # TODO: Can we test whether something is contained within a requirement?
80	    #       If so how do we do that? Do we need to test against the _name_ of
81	    #       the thing as well as the version? What about the markers?
82	    # TODO: Can we normalize the name and extra name?
83	
84	    __slots__ = ("extras", "marker", "name", "specifier", "url")
85	
86	    def __init__(self, requirement_string: str) -> None:
87	        try:
88	            parsed = _parse_requirement(requirement_string)
89	        except ParserSyntaxError as e:
90	            raise InvalidRequirement(str(e)) from e
91	
92	        self.name: str = parsed.name
93	        """The name of the requirement."""
94	        self.url: str | None = parsed.url or None
95	        """The URL to download the requirement from, or ``None``."""
96	        self.extras: set[str] = set(parsed.extras)
97	        """A set of extras that the requirement specifies."""
98	        try:
99	            self.specifier: SpecifierSet = SpecifierSet(parsed.specifier)
100	            """A :class:`~.SpecifierSet` of the version specified by the requirement."""
101	        except InvalidSpecifier as e:
102	            raise InvalidRequirement(str(e)) from e
103	        self.marker: Marker | None = None
104	        """A :class:`~.Marker` for the requirement, or ``None``."""
105	        if parsed.marker is not None:
106	            self.marker = Marker.__new__(Marker)
107	            self.marker._markers = _normalize_extra_values(parsed.marker)
108	
109	    def _iter_parts(self, name: str) -> Iterator[str]:
110	        yield name
111	
112	        if self.extras:
113	            formatted_extras = ",".join(sorted(self.extras))
114	            yield f"[{formatted_extras}]"
115	
116	        if self.specifier:
117	            yield str(self.specifier)
118	
119	        if self.url:
120	            yield f" @ {self.url}"
121	            if self.marker:
122	                yield " "
123	
124	        if self.marker:
125	            yield f"; {self.marker}"
126	
127	    def __getstate__(self) -> tuple[str, bool | None]:
128	        # Return the requirement string for compactness and stability, paired
129	        # with the specifier's explicit prereleases override, which is not
130	        # captured by the string form. Re-parsed on load to reconstruct all
131	        # other fields.
132	        return (str(self), self.specifier._prereleases)
133	
134	    def __setstate__(self, state: object) -> None:
135	        if isinstance(state, str):
136	            # Format (26.2): just the requirement string.
137	            requirement_string: str = state
138	            prereleases: bool | None = None
139	        elif (
140	            isinstance(state, tuple)
141	            and len(state) == 2
142	            and isinstance(state[0], str)
143	            and (state[1] is None or isinstance(state[1], bool))
144	        ):
145	            # New format (26.3+): (requirement string, specifier prereleases).
146	            requirement_string, prereleases = state
147	        elif isinstance(state, dict) and state.keys() >= set(self.__slots__):
148	            # Old format (packaging <= 26.1, no __slots__): plain __dict__.
149	            for key in self.__slots__:
150	                setattr(self, key, state[key])
151	            return
152	        else:
153	            raise TypeError(f"Cannot restore Requirement from {state!r}")
154	
155	        try:
156	            tmp = Requirement(requirement_string)
157	        except InvalidRequirement as exc:
158	            raise TypeError(f"Cannot restore Requirement from {state!r}") from exc
159	        self.name = tmp.name
160	        self.url = tmp.url
161	        self.extras = tmp.extras
162	        self.specifier = tmp.specifier
163	        self.specifier._prereleases = prereleases
164	        self.marker = tmp.marker
165	
166	    def __str__(self) -> str:
167	        return "".join(self._iter_parts(self.name))
168	
169	    def __repr__(self) -> str:
170	        return f"<{self.__class__.__name__}({str(self)!r})>"
171	
172	    def __hash__(self) -> int:
173	        # Mirror __eq__ by hashing the canonical specifier object rather than
174	        # its raw string. ``_iter_parts`` yields ``str(self.specifier)``, which
175	        # is non-canonical, so trailing-zero-equivalent requirements such as
176	        # ``foo==1.0.0`` and ``foo==1.0.0.0`` (which compare equal) would
177	        # otherwise hash differently, breaking the hash/__eq__ invariant.
178	        return hash(
179	            (
180	                canonicalize_name(self.name),
181	                frozenset(canonicalize_name(e) for e in self.extras),
182	                self.specifier,
183	                self.url,
184	                self.marker,
185	            )
186	        )
187	
188	    def __eq__(self, other: object) -> bool:
189	        if not isinstance(other, Requirement):
190	            return NotImplemented
191	
192	        # Extras must be normalized before comparison as per PEP 685.
193	        self_extras = frozenset(canonicalize_name(e) for e in self.extras)
194	        other_extras = frozenset(canonicalize_name(e) for e in other.extras)
195	        return (
196	            canonicalize_name(self.name) == canonicalize_name(other.name)
197	            and self_extras == other_extras
198	            and self.specifier == other.specifier
199	            and self.url == other.url
200	            and self.marker == other.marker
201	        )
