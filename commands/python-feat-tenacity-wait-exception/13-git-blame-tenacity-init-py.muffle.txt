141	class BaseAction:
142	    """Base class for representing actions to take by retry object.
143	
144	    Concrete implementations must define:
145	    - __init__: to initialize all necessary fields
146	    - REPR_FIELDS: class variable specifying attributes to include in repr(self)
147	    - NAME: for identification in retry object methods and callbacks
148	    """
149	
150	    REPR_FIELDS: t.Sequence[str] = ()
151	    NAME: t.Optional[str] = None
152	
153	    def __repr__(self) -> str:
154	        state_str = ", ".join(
155	            f"{field}={getattr(self, field)!r}" for field in self.REPR_FIELDS
156	        )
157	        return f"{self.__class__.__name__}({state_str})"
158	
159	    def __str__(self) -> str:
160	        return repr(self)

178	class RetryError(Exception):
179	    """Encapsulates the last attempt instance right before giving up."""
180	
181	    def __init__(self, last_attempt: "Future") -> None:
182	        self.last_attempt = last_attempt
183	        super().__init__(last_attempt)
184	
185	    def reraise(self) -> t.NoReturn:
186	        if self.last_attempt.failed:
187	            raise self.last_attempt.result()
188	        raise self
189	
190	    def __str__(self) -> str:
191	        return f"{self.__class__.__name__}[{self.last_attempt}]"

194	class AttemptManager:
195	    """Manage attempt context."""
196	
197	    def __init__(self, retry_state: "RetryCallState"):
198	        self.retry_state = retry_state
199	
200	    def __enter__(self) -> None:

202	
203	    def __exit__(
204	        self,
205	        exc_type: t.Optional[t.Type[BaseException]],
206	        exc_value: t.Optional[BaseException],
207	        traceback: t.Optional["types.TracebackType"],
208	    ) -> t.Optional[bool]:
209	        if exc_type is not None and exc_value is not None:
210	            self.retry_state.set_exception((exc_type, exc_value, traceback))
211	            return True  # Swallow exception.
212	        else:
213	            # We don't have the result, actually.
214	            self.retry_state.set_result(None)
215	            return None

288	    @property
289	    def statistics(self) -> t.Dict[str, t.Any]:
290	        """Return a dictionary of runtime statistics.
291	
292	        This dictionary will be empty when the controller has never been
293	        ran. When it is running or has ran previously it should have (but
294	        may not) have useful and/or informational keys and values when
295	        running is underway and/or completed.
296	
297	        .. warning:: The keys in this dictionary **should** be some what
298	                     stable (not changing), but there existence **may**
299	                     change between major releases as new statistics are
300	                     gathered or removed so before accessing keys ensure that
301	                     they actually exist and handle when they do not.
302	
303	        .. note:: The values in this dictionary are local to the thread
304	                  running call (so if multiple threads share the same retrying
305	                  object - either directly or indirectly) they will each have
306	                  there own view of statistics they have collected (in the
307	                  future we may provide a way to aggregate the various
308	                  statistics from each thread).
309	        """
310	        if not hasattr(self._local, "statistics"):
311	            self._local.statistics = t.cast(t.Dict[str, t.Any], {})
312	        return self._local.statistics  # type: ignore[no-any-return]
313	
314	    @property
315	    def iter_state(self) -> IterState:
316	        if not hasattr(self._local, "iter_state"):
317	            self._local.iter_state = IterState()
318	        return self._local.iter_state  # type: ignore[no-any-return]

320	    def wraps(self, f: WrappedFn) -> WrappedFn:
321	        """Wrap a function for retrying.
322	
323	        :param f: A function to wraps for retrying.
324	        """
325	
326	        @functools.wraps(
327	            f, functools.WRAPPER_ASSIGNMENTS + ("__defaults__", "__kwdefaults__")
328	        )
329	        def wrapped_f(*args: t.Any, **kw: t.Any) -> t.Any:
330	            # Always create a copy to prevent overwriting the local contexts when
331	            # calling the same wrapped functions multiple times in the same stack
332	            copy = self.copy()
333	            wrapped_f.statistics = copy.statistics  # type: ignore[attr-defined]
334	            return copy(f, *args, **kw)
335	
336	        def retry_with(*args: t.Any, **kwargs: t.Any) -> WrappedFn:
337	            return self.copy(*args, **kwargs).wraps(f)
338	
339	        # Preserve attributes
340	        wrapped_f.retry = self  # type: ignore[attr-defined]
341	        wrapped_f.retry_with = retry_with  # type: ignore[attr-defined]
342	        wrapped_f.statistics = {}  # type: ignore[attr-defined]
343	
344	        return wrapped_f  # type: ignore[return-value]
345	
346	    def begin(self) -> None:
347	        self.statistics.clear()
348	        self.statistics["start_time"] = time.monotonic()
349	        self.statistics["attempt_number"] = 1
350	        self.statistics["idle_for"] = 0
351	
352	    def _add_action_func(self, fn: t.Callable[..., t.Any]) -> None:
353	        self.iter_state.actions.append(fn)
354	
355	    def _run_retry(self, retry_state: "RetryCallState") -> None:
356	        self.iter_state.retry_run_result = self.retry(retry_state)
357	
358	    def _run_wait(self, retry_state: "RetryCallState") -> None:
359	        if self.wait:
360	            sleep = self.wait(retry_state)
361	        else:
362	            sleep = 0.0
363	
364	        retry_state.upcoming_sleep = sleep
365	
366	    def _run_stop(self, retry_state: "RetryCallState") -> None:
367	        self.statistics["delay_since_first_attempt"] = retry_state.seconds_since_start
368	        self.iter_state.stop_run_result = self.stop(retry_state)
369	
370	    def iter(self, retry_state: "RetryCallState") -> t.Union[DoAttempt, DoSleep, t.Any]:  # noqa
371	        self._begin_iter(retry_state)
372	        result = None
373	        for action in self.iter_state.actions:
374	            result = action(retry_state)
375	        return result
376	
377	    def _begin_iter(self, retry_state: "RetryCallState") -> None:  # noqa
378	        self.iter_state.reset()
379	
380	        fut = retry_state.outcome
381	        if fut is None:
382	            if self.before is not None:
383	                self._add_action_func(self.before)
384	            self._add_action_func(lambda rs: DoAttempt())
385	            return
386	
387	        self.iter_state.is_explicit_retry = fut.failed and isinstance(
388	            fut.exception(), TryAgain
389	        )
390	        if not self.iter_state.is_explicit_retry:
391	            self._add_action_func(self._run_retry)
392	        self._add_action_func(self._post_retry_check_actions)
393	
394	    def _post_retry_check_actions(self, retry_state: "RetryCallState") -> None:
395	        if not (self.iter_state.is_explicit_retry or self.iter_state.retry_run_result):
396	            self._add_action_func(lambda rs: rs.outcome.result())
397	            return
398	
399	        if self.after is not None:
400	            self._add_action_func(self.after)
401	
402	        self._add_action_func(self._run_wait)
403	        self._add_action_func(self._run_stop)
404	        self._add_action_func(self._post_stop_check_actions)
405	
406	    def _post_stop_check_actions(self, retry_state: "RetryCallState") -> None:
407	        if self.iter_state.stop_run_result:
408	            if self.retry_error_callback:
409	                self._add_action_func(self.retry_error_callback)
410	                return
411	
412	            def exc_check(rs: "RetryCallState") -> None:
413	                fut = t.cast(Future, rs.outcome)
414	                retry_exc = self.retry_error_cls(fut)
415	                if self.reraise:
416	                    raise retry_exc.reraise()
417	                raise retry_exc from fut.exception()
418	
419	            self._add_action_func(exc_check)
420	            return
421	
422	        def next_action(rs: "RetryCallState") -> None:
423	            sleep = rs.upcoming_sleep
424	            rs.next_action = RetryAction(sleep)
425	            rs.idle_for += sleep
426	            self.statistics["idle_for"] += sleep
427	            self.statistics["attempt_number"] += 1
428	
429	        self._add_action_func(next_action)
430	
431	        if self.before_sleep is not None:
432	            self._add_action_func(self.before_sleep)
433	
434	        self._add_action_func(lambda rs: DoSleep(rs.upcoming_sleep))
435	
436	    def __iter__(self) -> t.Generator[AttemptManager, None, None]:
437	        self.begin()
438	
439	        retry_state = RetryCallState(self, fn=None, args=(), kwargs={})
440	        while True:
441	            do = self.iter(retry_state=retry_state)
442	            if isinstance(do, DoAttempt):
443	                yield AttemptManager(retry_state=retry_state)
444	            elif isinstance(do, DoSleep):
445	                retry_state.prepare_for_next_attempt()
446	                self.sleep(do)
447	            else:
448	                break

460	class Retrying(BaseRetrying):
461	    """Retrying controller."""
462	
463	    def __call__(
464	        self,
465	        fn: t.Callable[..., WrappedFnReturnT],
466	        *args: t.Any,
467	        **kwargs: t.Any,
468	    ) -> WrappedFnReturnT:
469	        self.begin()
470	
471	        retry_state = RetryCallState(retry_object=self, fn=fn, args=args, kwargs=kwargs)
472	        while True:
473	            do = self.iter(retry_state=retry_state)
474	            if isinstance(do, DoAttempt):
475	                try:
476	                    result = fn(*args, **kwargs)
477	                except BaseException:  # noqa: B902
478	                    retry_state.set_exception(sys.exc_info())  # type: ignore[arg-type]
479	                else:
480	                    retry_state.set_result(result)
481	            elif isinstance(do, DoSleep):
482	                retry_state.prepare_for_next_attempt()
483	                self.sleep(do)
484	            else:
485	                return do  # type: ignore[no-any-return]

494	class Future(FutureGenericT):
495	    """Encapsulates a (future or past) attempted call to a target function."""
496	
497	    def __init__(self, attempt_number: int) -> None:
498	        super().__init__()
499	        self.attempt_number = attempt_number
500	
501	    @property
502	    def failed(self) -> bool:
503	        """Return whether a exception is being held in this future."""
504	        return self.exception() is not None
505	
506	    @classmethod
507	    def construct(
508	        cls, attempt_number: int, value: t.Any, has_exception: bool
509	    ) -> "Future":
510	        """Construct a new Future object."""
511	        fut = cls(attempt_number)
512	        if has_exception:
513	            fut.set_exception(value)
514	        else:
515	            fut.set_result(value)
516	        return fut

519	class RetryCallState:
520	    """State related to a single call wrapped with Retrying."""
521	
522	    def __init__(
523	        self,
524	        retry_object: BaseRetrying,
525	        fn: t.Optional[WrappedFn],
526	        args: t.Any,
527	        kwargs: t.Any,
528	    ) -> None:
529	        #: Retry call start timestamp
530	        self.start_time = time.monotonic()
531	        #: Retry manager object
532	        self.retry_object = retry_object
533	        #: Function wrapped by this retry call
534	        self.fn = fn
535	        #: Arguments of the function wrapped by this retry call
536	        self.args = args
537	        #: Keyword arguments of the function wrapped by this retry call
538	        self.kwargs = kwargs
539	
540	        #: The number of the current attempt
541	        self.attempt_number: int = 1
542	        #: Last outcome (result or exception) produced by the function
543	        self.outcome: t.Optional[Future] = None
544	        #: Timestamp of the last outcome
545	        self.outcome_timestamp: t.Optional[float] = None
546	        #: Time spent sleeping in retries
547	        self.idle_for: float = 0.0
548	        #: Next action as decided by the retry manager
549	        self.next_action: t.Optional[RetryAction] = None
550	        #: Next sleep time as decided by the retry manager.
551	        self.upcoming_sleep: float = 0.0
552	
553	    @property
554	    def seconds_since_start(self) -> t.Optional[float]:
555	        if self.outcome_timestamp is None:
556	            return None
557	        return self.outcome_timestamp - self.start_time
558	
559	    def prepare_for_next_attempt(self) -> None:
560	        self.outcome = None
561	        self.outcome_timestamp = None
562	        self.attempt_number += 1
563	        self.next_action = None
564	
565	    def set_result(self, val: t.Any) -> None:
566	        ts = time.monotonic()
567	        fut = Future(self.attempt_number)
568	        fut.set_result(val)
569	        self.outcome, self.outcome_timestamp = fut, ts
570	
571	    def set_exception(
572	        self,
573	        exc_info: t.Tuple[
574	            t.Type[BaseException], BaseException, "types.TracebackType| None"
575	        ],
576	    ) -> None:
577	        ts = time.monotonic()
578	        fut = Future(self.attempt_number)
579	        fut.set_exception(exc_info[1])
580	        self.outcome, self.outcome_timestamp = fut, ts
581	
582	    def __repr__(self) -> str:
583	        if self.outcome is None:
584	            result = "none yet"
585	        elif self.outcome.failed:
586	            exception = self.outcome.exception()
587	            result = f"failed ({exception.__class__.__name__} {exception})"
588	        else:
589	            result = f"returned {self.outcome.result()}"
590	
591	        slept = float(round(self.idle_for, 2))
592	        clsname = self.__class__.__name__
593	        return f"<{clsname} {id(self)}: attempt #{self.attempt_number}; slept for {slept}; last result: {result}>"

627	    :param dkw: keyword arguments passed to the Retrying object

632	    else:

634	        def wrap(f: WrappedFn) -> WrappedFn:
635	            if isinstance(f, retry_base):
636	                warnings.warn(

638	                    f"this will probably hang indefinitely (did you mean retry={f.__class__.__name__}(...)?)"

649	            else:
