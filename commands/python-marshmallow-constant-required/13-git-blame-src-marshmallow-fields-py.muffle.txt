101	def _resolve_field_instance(cls_or_instance: Field | type[Field]) -> Field:
102	    """Return a Field instance from a Field class or instance.
103	
104	    :param cls_or_instance: Field class or instance.
105	    """
106	    if isinstance(cls_or_instance, type):

115	class Field(typing.Generic[_InternalT]):
116	    """Base field from which all other fields inherit.
117	    This class should not be used directly within Schemas.
118	
119	    :param dump_default: If set, this value will be used during serialization if the
120	        input value is missing. If not set, the field will be excluded from the
121	        serialized output if the input value is missing. May be a value or a callable.
122	    :param load_default: Default deserialization value for the field if the field is not
123	        found in the input data. May be a value or a callable.
124	    :param data_key: The name of the dict key in the external representation, i.e.
125	        the input of `load` and the output of `dump`.
126	        If `None`, the key will match the name of the field.
127	    :param attribute: The name of the key/attribute in the internal representation, i.e.
128	        the output of `load` and the input of `dump`.
129	        If `None`, the key/attribute will match the name of the field.
130	        Note: This should only be used for very specific use cases such as
131	        outputting multiple fields for a single attribute, or using keys/attributes
132	        that are invalid variable names, unsuitable for field names. In most cases,
133	        you should use ``data_key`` instead.
134	    :param validate: Validator or collection of validators that are called
135	        during deserialization. Validator takes a field's input value as
136	        its only parameter and returns a boolean.
137	        If it returns `False`, an :exc:`ValidationError` is raised.
138	    :param required: Raise a :exc:`ValidationError` if the field value
139	        is not supplied during deserialization.
140	    :param allow_none: Set this to `True` if `None` should be considered a valid value during
141	        validation/deserialization. If set to `False` (the default), `None` is considered invalid input.
142	        If ``load_default`` is explicitly set to `None` and ``allow_none`` is unset,
143	        `allow_none` is implicitly set to ``True``.
144	    :param load_only: If `True` skip this field during serialization, otherwise
145	        its value will be present in the serialized data.
146	    :param dump_only: If `True` skip this field during deserialization, otherwise
147	        its value will be present in the deserialized object. In the context of an
148	        HTTP API, this effectively marks the field as "read-only".
149	    :param error_messages: Overrides for `Field.default_error_messages`.
150	    :param metadata: Extra information to be stored as field metadata.
151	
152	    .. versionchanged:: 3.0.0b8
153	        Add ``data_key`` parameter for the specifying the key in the input and
154	        output data. This parameter replaced both ``load_from`` and ``dump_to``.
155	    .. versionchanged:: 3.13.0
156	        Replace ``missing`` and ``default`` parameters with ``load_default`` and ``dump_default``.
157	    .. versionchanged:: 3.24.0
158	        `Field <marshmallow.fields.Field>` should no longer be used as a field within a `Schema <marshmallow.Schema>`.
159	        Use `Raw <marshmallow.fields.Raw>` or another `Field <marshmallow.fields.Field>` subclass instead.
160	    .. versionchanged:: 4.0.0
161	        Remove ``context`` property.
162	    """

178	    def __init__(
179	        self,
180	        *,
181	        load_default: typing.Any = missing_,
182	        dump_default: typing.Any = missing_,
183	        data_key: str | None = None,
184	        attribute: str | None = None,
185	        validate: types.Validator | typing.Iterable[types.Validator] | None = None,
186	        required: bool = False,
187	        allow_none: bool | None = None,
188	        load_only: bool = False,
189	        dump_only: bool = False,
190	        error_messages: dict[str, str] | None = None,
191	        metadata: typing.Mapping[str, typing.Any] | None = None,
192	    ) -> None:
193	        self.dump_default = dump_default
194	        self.load_default = load_default
195	
196	        self.attribute = attribute
197	        self.data_key = data_key
198	        self.validate = validate
199	        if validate is None:
200	            self.validators = []
201	        elif callable(validate):
202	            self.validators = [validate]
203	        elif utils.is_iterable_but_not_string(validate):
204	            self.validators = list(validate)
205	        else:
206	            raise ValueError(
207	                "The 'validate' parameter must be a callable "
208	                "or a collection of callables."
209	            )
210	
211	        # If allow_none is None and load_default is None
212	        # None should be considered valid by default
213	        self.allow_none = load_default is None if allow_none is None else allow_none
214	        self.load_only = load_only
215	        self.dump_only = dump_only
216	        if required is True and load_default is not missing_:
217	            raise ValueError("'load_default' must not be set for required fields.")
218	        self.required = required
219	
220	        metadata = metadata or {}
221	        self.metadata = metadata
222	        # Collect default error message from self and parent classes
223	        messages: dict[str, str] = {}
224	        for cls in reversed(self.__class__.__mro__):
225	            messages.update(getattr(cls, "default_error_messages", {}))
226	        messages.update(error_messages or {})
227	        self.error_messages = messages

229	        self.parent: Field | Schema | None = None
230	        self.name: str | None = None
231	        self.root: Schema | None = None

276	    def make_error(self, key: str, **kwargs) -> ValidationError:
277	        """Helper method to make a `ValidationError` with an error message
278	        from ``self.error_messages``.
279	        """
280	        try:
281	            msg = self.error_messages[key]
282	        except KeyError as error:
283	            class_name = self.__class__.__name__
284	            message = (
285	                f"ValidationError raised by `{class_name}`, but error key `{key}` does "
286	                "not exist in the `error_messages` dictionary."
287	            )
288	            raise AssertionError(message) from error
289	        if isinstance(msg, (str, bytes)):
290	            msg = msg.format(**kwargs)
291	        return ValidationError(msg)

293	    def _validate_missing(self, value: typing.Any) -> None:
294	        """Validate missing values. Raise a :exc:`ValidationError` if
295	        `value` should be considered missing.
296	        """
297	        if value is missing_ and self.required:
298	            raise self.make_error("required")
299	        if value is None and not self.allow_none:
300	            raise self.make_error("null")

302	    def serialize(
303	        self,
304	        attr: str,
305	        obj: typing.Any,
306	        accessor: (
307	            typing.Callable[[typing.Any, str, typing.Any], typing.Any] | None
308	        ) = None,
309	        **kwargs,
310	    ):
311	        """Pulls the value for the given key from the object, applies the
312	        field's formatting and returns the result.
313	
314	        :param attr: The attribute/key to get from the object.
315	        :param obj: The object to access the attribute/key from.
316	        :param accessor: Function used to access values from ``obj``.
317	        :param kwargs: Field-specific keyword arguments.
318	        """
319	        if self._CHECK_ATTRIBUTE:
320	            value = self.get_value(obj, attr, accessor=accessor)
321	            if value is missing_:
322	                default = self.dump_default
323	                value = default() if callable(default) else default
324	            if value is missing_:
325	                return value
326	        else:
327	            value = None
328	        return self._serialize(value, attr, obj, **kwargs)

350	    def deserialize(
351	        self,
352	        value: typing.Any,
353	        attr: str | None = None,
354	        data: typing.Mapping[str, typing.Any] | None = None,
355	        **kwargs,
356	    ) -> _InternalT | None:
357	        """Deserialize ``value``.
358	
359	        :param value: The value to deserialize.
360	        :param attr: The attribute/key in `data` to deserialize.
361	        :param data: The raw input data passed to `Schema.load <marshmallow.Schema.load>`.
362	        :param kwargs: Field-specific keyword arguments.
363	        :raise ValidationError: If an invalid value is passed or if a required value
364	            is missing.
365	        """
366	        # Validate required fields, deserialize, then validate
367	        # deserialized value
368	        self._validate_missing(value)
369	        if value is missing_:
370	            _miss = self.load_default
371	            return _miss() if callable(_miss) else _miss
372	        if self.allow_none and value is None:
373	            return None
374	        output = self._deserialize(value, attr, data, **kwargs)
375	        self._validate(output)
376	        return output

441	class Nested(Field):
442	    """Allows you to nest a :class:`Schema <marshmallow.Schema>`
443	    inside a field.
444	
445	    Examples: ::
446	
447	        class ChildSchema(Schema):
448	            id = fields.Str()
449	            name = fields.Str()
450	            # Use lambda functions when you need two-way nesting or self-nesting
451	            parent = fields.Nested(lambda: ParentSchema(only=("id",)), dump_only=True)
452	            siblings = fields.List(
453	                fields.Nested(lambda: ChildSchema(only=("id", "name")))
454	            )
455	
456	
457	        class ParentSchema(Schema):
458	            id = fields.Str()
459	            children = fields.List(
460	                fields.Nested(ChildSchema(only=("id", "parent", "siblings")))
461	            )
462	            spouse = fields.Nested(lambda: ParentSchema(only=("id",)))
463	
464	    When passing a `Schema <marshmallow.Schema>` instance as the first argument,
465	    the instance's ``exclude``, ``only``, and ``many`` attributes will be respected.
466	
467	    Therefore, when passing the ``exclude``, ``only``, or ``many`` arguments to `fields.Nested`,
468	    you should pass a `Schema <marshmallow.Schema>` class (not an instance) as the first argument.
469	
470	    ::
471	
472	        # Yes
473	        author = fields.Nested(UserSchema, only=("id", "name"))
474	
475	        # No
476	        author = fields.Nested(UserSchema(), only=("id", "name"))
477	
478	    :param nested: `Schema <marshmallow.Schema>` instance, class, class name (string), dictionary, or callable that
479	        returns a `Schema <marshmallow.Schema>` or dictionary.
480	        Dictionaries are converted with `Schema.from_dict <marshmallow.Schema.from_dict>`.
481	    :param exclude: A list or tuple of fields to exclude.
482	    :param only: A list or tuple of fields to marshal. If `None`, all fields are marshalled.
483	        This parameter takes precedence over ``exclude``.
484	    :param many: Whether the field is a collection of objects.
485	        If `None` (default), and nested `Schema` instance is provided, ``many`` is not overridden.
486	        If `None` (default), and `Schema` subclass is provided, schema instance sets ``many`` as False.
487	        If `True | False` nested `[nested_field].schema.many` is overridden.
488	    :param unknown: Whether to exclude, include, or raise an error for unknown
489	        fields in the data. Use `EXCLUDE`, `INCLUDE` or `RAISE`.
490	    :param kwargs: The same keyword arguments that :class:`Field` receives.
491	    """
492	
493	    #: Default error messages.
494	    default_error_messages = {"type": "Invalid type."}
495	
496	    def __init__(
497	        self,
498	        nested: (
499	            Schema
500	            | SchemaMeta
501	            | str
502	            | dict[str, Field]
503	            | typing.Callable[[], Schema | SchemaMeta | dict[str, Field]]
504	        ),
505	        *,
506	        only: types.StrSequenceOrSet | None = None,
507	        exclude: types.StrSequenceOrSet = (),
508	        many: bool | None = None,
509	        unknown: types.UnknownOption | None = None,
510	        **kwargs: Unpack[_BaseFieldKwargs],
511	    ):
512	        # Raise error if only or exclude is passed as string, not list of strings
513	        if only is not None and not utils.is_sequence_but_not_string(only):
514	            raise StringNotCollectionError('"only" should be a collection of strings.')
515	        if not utils.is_sequence_but_not_string(exclude):
516	            raise StringNotCollectionError(
517	                '"exclude" should be a collection of strings.'
518	            )
519	        self.nested = nested
520	        self.only = only
521	        self.exclude = exclude
522	        self.many = many
523	        self.unknown = unknown
524	        self._schema: Schema | None = None  # Cached Schema instance
525	        super().__init__(**kwargs)
526	
527	    @property
528	    def schema(self) -> Schema:
529	        """The nested Schema object."""
530	        if not self._schema:
531	            if callable(self.nested) and not isinstance(self.nested, type):
532	                nested = self.nested()
533	            else:
534	                nested = typing.cast("Schema", self.nested)
535	            # defer the import of `marshmallow.schema` to avoid circular imports
536	            from marshmallow.schema import Schema  # noqa: PLC0415
537	
538	            if isinstance(nested, dict):
539	                nested = Schema.from_dict(nested)
540	
541	            if isinstance(nested, Schema):
542	                self._schema = copy.copy(nested)
543	                # Respect only and exclude and many passed from parent and re-initialize fields
544	                set_class = typing.cast("type[set]", self._schema.set_class)
545	                if self.only is not None:
546	                    if self._schema.only is not None:
547	                        original = self._schema.only
548	                    else:  # only=None -> all fields
549	                        original = self._schema.fields.keys()
550	                    self._schema.only = set_class(self.only) & set_class(original)
551	                if self.exclude:
552	                    original = self._schema.exclude
553	                    self._schema.exclude = set_class(self.exclude) | set_class(original)
554	                if self.many is not None:
555	                    self._schema.many = self.many
556	                self._schema._init_fields()
557	            else:
558	                if isinstance(nested, type) and issubclass(nested, Schema):
559	                    schema_class: type[Schema] = nested
560	                elif not isinstance(nested, (str, bytes)):
561	                    raise ValueError(
562	                        "`Nested` fields must be passed a "
563	                        f"`Schema`, not {nested.__class__}."
564	                    )
565	                else:
566	                    schema_class = class_registry.get_class(nested, all=False)
567	                self._schema = schema_class(
568	                    many=bool(self.many),
569	                    only=self.only,
570	                    exclude=self.exclude,
571	                    load_only=self._nested_normalized_option("load_only"),
572	                    dump_only=self._nested_normalized_option("dump_only"),
573	                )
574	        return self._schema
575	
576	    def _nested_normalized_option(self, option_name: str) -> list[str]:
577	        nested_field = f"{self.name}."
578	        return [
579	            field.split(nested_field, 1)[1]
580	            for field in getattr(self.root, option_name, set())
581	            if field.startswith(nested_field)
582	        ]

584	    def _serialize(self, nested_obj, attr, obj, **kwargs):
585	        # Load up the schema first. This allows a RegistryError to be raised
586	        # if an invalid schema name was passed
587	        schema = self.schema
588	        if nested_obj is None:
589	            return None
590	        many = schema.many or self.many
591	        return schema.dump(nested_obj, many=many)

593	    def _test_collection(self, value: typing.Any) -> None:
594	        many = self.schema.many or self.many
595	        if many and not utils.is_collection(value):
596	            raise self.make_error("type", input=value, type=value.__class__.__name__)

598	    def _load(
599	        self,
600	        value: typing.Any,
601	        partial: bool | types.StrSequenceOrSet | None = None,  # noqa: FBT001
602	    ):
603	        try:
604	            valid_data = self.schema.load(value, unknown=self.unknown, partial=partial)
605	        except ValidationError as error:
606	            raise ValidationError(
607	                error.messages, valid_data=error.valid_data
608	            ) from error
609	        return valid_data

612	        self,
613	        value: typing.Any,
614	        attr: str | None,
615	        data: typing.Mapping[str, typing.Any] | None,
616	        partial: bool | types.StrSequenceOrSet | None = None,  # noqa: FBT001

631	class Pluck(Nested):
632	    """Allows you to replace nested data with one of the data's fields.
633	
634	    Example: ::
635	
636	        from marshmallow import Schema, fields
637	
638	
639	        class ArtistSchema(Schema):
640	            id = fields.Int()
641	            name = fields.Str()
642	
643	
644	        class AlbumSchema(Schema):
645	            artist = fields.Pluck(ArtistSchema, "id")
646	
647	
648	        in_data = {"artist": 42}
649	        loaded = AlbumSchema().load(in_data)  # => {'artist': {'id': 42}}
650	        dumped = AlbumSchema().dump(loaded)  # => {'artist': 42}
651	
652	    :param nested: The Schema class or class name (string) to nest
653	    :param str field_name: The key to pluck a value from.
654	    :param kwargs: The same keyword arguments that :class:`Nested` receives.
655	    """
656	
657	    def __init__(
658	        self,
659	        nested: Schema | SchemaMeta | str | typing.Callable[[], Schema],
660	        field_name: str,
661	        *,
662	        many: bool = False,
663	        unknown: types.UnknownOption | None = None,
664	        **kwargs: Unpack[_BaseFieldKwargs],
665	    ):
666	        super().__init__(
667	            nested, only=(field_name,), many=many, unknown=unknown, **kwargs
668	        )
669	        self.field_name = field_name
670	
671	    @property
672	    def _field_data_key(self) -> str:
673	        only_field = self.schema.fields[self.field_name]
674	        return only_field.data_key or self.field_name
675	
676	    def _serialize(self, nested_obj, attr, obj, **kwargs):
677	        ret = super()._serialize(nested_obj, attr, obj, **kwargs)
678	        if ret is None:
679	            return None
680	        if self.many:
681	            return utils.pluck(ret, key=self._field_data_key)
682	        return ret[self._field_data_key]
683	
684	    def _deserialize(self, value, attr, data, partial=None, **kwargs):
685	        self._test_collection(value)
686	        if self.many:
687	            value = [{self._field_data_key: v} for v in value]
688	        else:
689	            value = {self._field_data_key: value}
690	        return self._load(value, partial=partial)

742	        if not utils.is_collection(value):

746	        errors = {}
747	        for idx, each in enumerate(value):

761	    instances
762	
763	    Example: ::
764	
765	        row = Tuple((fields.String(), fields.Integer(), fields.Float()))
766	
767	    .. note::
768	        Because of the structured nature of `collections.namedtuple` and
769	        `typing.NamedTuple`, using a Schema within a Nested field for them is
770	        more appropriate than using a `Tuple` field.
771	
772	    :param tuple_fields: An iterable of field classes or
773	        instances.
774	    :param kwargs: The same keyword arguments that :class:`Field` receives.
775	
776	    .. versionadded:: 3.0.0rc4
777	    """
778	
779	    #: Default error messages.
780	    default_error_messages = {"invalid": "Not a valid tuple."}
781	
782	    def __init__(
783	        self,
784	        tuple_fields: typing.Iterable[Field | type[Field]],
785	        **kwargs: Unpack[_BaseFieldKwargs],
786	    ):
787	        super().__init__(**kwargs)
788	        if not utils.is_collection(tuple_fields):
789	            raise ValueError(
790	                "tuple_fields must be an iterable of Field classes or instances."
791	            )
792	
793	        try:
794	            self.tuple_fields = [
795	                _resolve_field_instance(cls_or_instance)
796	                for cls_or_instance in tuple_fields
797	            ]
798	        except _FieldInstanceResolutionError as error:
799	            raise ValueError(
800	                'Elements of "tuple_fields" must be subclasses or '
801	                "instances of marshmallow.fields.Field."
802	            ) from error
803	
804	        self.validate_length = Length(equal=len(self.tuple_fields))
805	
806	    def _bind_to_schema(self, field_name: str, parent: Schema | Field) -> None:
807	        super()._bind_to_schema(field_name, parent)
808	        new_tuple_fields = []
809	        for field in self.tuple_fields:
810	            new_field = copy.deepcopy(field)
811	            new_field._bind_to_schema(field_name, self)
812	            new_tuple_fields.append(new_field)
813	
814	        self.tuple_fields = new_tuple_fields
815	
816	    def _serialize(
817	        self, value: tuple | None, attr: str | None, obj: typing.Any, **kwargs
818	    ) -> tuple | None:
819	        if value is None:
820	            return None
821	
822	        return tuple(
823	            field._serialize(each, attr, obj, **kwargs)
824	            for field, each in zip(self.tuple_fields, value, strict=True)
825	        )
826	
827	    def _deserialize(
828	        self,
829	        value: typing.Any,
830	        attr: str | None,
831	        data: typing.Mapping[str, typing.Any] | None,
832	        **kwargs,
833	    ) -> tuple:
834	        if not utils.is_sequence_but_not_string(value):
835	            raise self.make_error("invalid")
836	
837	        self.validate_length(value)
838	
839	        result = []
840	        errors = {}
841	
842	        for idx, (field, each) in enumerate(zip(self.tuple_fields, value, strict=True)):
843	            try:
844	                result.append(field.deserialize(each, **kwargs))
845	            except ValidationError as error:
846	                if error.valid_data is not None:
847	                    result.append(error.valid_data)
848	                errors.update({idx: error.messages})
849	        if errors:
850	            raise ValidationError(errors, valid_data=result)
851	
852	        return tuple(result)

910	class Number(Field[_NumT], metaclass=abc.ABCMeta):
911	    """Abstract base class for number fields.
912	
913	    Use `Integer <marshmallow.fields.Integer>`, `Float <marshmallow.fields.Float>`,
914	    or `Decimal <marshmallow.fields.Decimal>` within schemas.
915	
916	    :param as_string: If `True`, format the serialized value as a string.
917	    :param kwargs: The same keyword arguments that :class:`Field` receives.
918	
919	    .. versionchanged:: 3.24.0
920	        `Number <marshmallow.fields.Number>` should no longer be used as a field within a `Schema <marshmallow.Schema>`.
921	        Use `Integer <marshmallow.fields.Integer>`, `Float <marshmallow.fields.Float>`, or `Decimal <marshmallow.fields.Decimal>` instead.
922	    """
923	
924	    @property
925	    @abc.abstractmethod
926	    def num_type(self) -> type[_NumT]: ...
927	
928	    #: Default error messages.
929	    default_error_messages = {
930	        "invalid": "Not a valid number.",
931	        "too_large": "Number too large.",
932	    }
933	
934	    def __init__(self, *, as_string: bool = False, **kwargs: Unpack[_BaseFieldKwargs]):
935	        self.as_string = as_string
936	        super().__init__(**kwargs)
937	
938	    def _format_num(self, value) -> _NumT:
939	        """Return the number value for value, given this field's `num_type`."""
940	        return self.num_type(value)  # type: ignore[call-arg]
941	
942	    def _validated(self, value: typing.Any) -> _NumT:
943	        """Format the value or raise a :exc:`ValidationError` if an error occurs."""
944	        # (value is True or value is False) is ~5x faster than isinstance(value, bool)
945	        if value is True or value is False:
946	            raise self.make_error("invalid", input=value)
947	        try:
948	            return self._format_num(value)
949	        except (TypeError, ValueError) as error:
950	            raise self.make_error("invalid", input=value) from error
951	        except OverflowError as error:
952	            raise self.make_error("too_large", input=value) from error
953	
954	    def _to_string(self, value: _NumT) -> str:
955	        return str(value)
956	
957	    def _serialize(self, value, attr, obj, **kwargs) -> str | _NumT | None:
958	        """Return a string if `self.as_string=True`, otherwise return this field's `num_type`."""
959	        if value is None:
960	            return None
961	        ret: _NumT = self._format_num(value)
962	        return self._to_string(ret) if self.as_string else ret
963	
964	    def _deserialize(self, value, attr, data, **kwargs) -> _NumT:
965	        return self._validated(value)

1032	class Decimal(Number[decimal.Decimal]):
1033	    """A field that (de)serializes to the Python ``decimal.Decimal`` type.
1034	    It's safe to use when dealing with money values, percentages, ratios
1035	    or other numbers where precision is critical.
1036	
1037	    .. warning::
1038	
1039	        This field serializes to a `decimal.Decimal` object by default. If you need
1040	        to render your data as JSON, keep in mind that the `json` module from the
1041	        standard library does not encode `decimal.Decimal`. Therefore, you must use
1042	        a JSON library that can handle decimals, such as `simplejson`, or serialize
1043	        to a string by passing ``as_string=True``.

1048	        first be cast to its corresponding `string` value before being deserialized
1049	        to a `decimal.Decimal` object. The default `__str__` implementation of the
1050	        built-in Python `float` type may apply a destructive transformation upon
1051	        its input data and therefore cannot be relied upon to preserve precision.
1052	        To avoid this, you can instead pass a JSON `string` to be deserialized
1053	        directly.
1054	
1055	    :param places: How many decimal places to quantize the value. If `None`, does
1056	        not quantize the value.
1057	    :param rounding: How to round the value during quantize, for example
1058	        `decimal.ROUND_UP`. If `None`, uses the rounding value from
1059	        the current thread's context.
1060	    :param allow_nan: If `True`, `NaN`, `Infinity` and `-Infinity` are allowed,
1061	        even though they are illegal according to the JSON specification.
1062	    :param as_string: If `True`, serialize to a string instead of a Python
1063	        `decimal.Decimal` type.
1064	    :param kwargs: The same keyword arguments that :class:`Number` receives.
1065	    """
1066	
1067	    num_type = decimal.Decimal
1068	
1069	    #: Default error messages.
1070	    default_error_messages = {
1071	        "special": "Special numeric values (nan or infinity) are not permitted."
1072	    }

1075	        self,

1081	        **kwargs: Unpack[_BaseFieldKwargs],
1082	    ):

1084	            decimal.Decimal((0, (1,), -places)) if places is not None else None

1086	        self.rounding = rounding
1087	        self.allow_nan = allow_nan
1088	        super().__init__(as_string=as_string, **kwargs)

1457	class TimeDelta(Field[dt.timedelta]):
1458	    """A field that (de)serializes a :class:`datetime.timedelta` object to a `float`.
1459	    The `float` can represent any time unit that the :class:`datetime.timedelta` constructor
1460	    supports.
1461	
1462	    :param precision: The time unit used for (de)serialization. Must be one of 'weeks',
1463	        'days', 'hours', 'minutes', 'seconds', 'milliseconds' or 'microseconds'.
1464	    :param kwargs: The same keyword arguments that :class:`Field` receives.
1465	
1466	    Float Caveats
1467	    -------------
1468	    Precision loss may occur when serializing a highly precise :class:`datetime.timedelta`
1469	    object using a big ``precision`` unit due to floating point arithmetics.
1470	
1471	    When necessary, the :class:`datetime.timedelta` constructor rounds `float` inputs
1472	    to whole microseconds during initialization of the object. As a result, deserializing
1473	    a `float` might be subject to rounding, regardless of `precision`. For example,
1474	    ``TimeDelta().deserialize("1.1234567") == timedelta(seconds=1, microseconds=123457)``.
1475	
1476	    .. versionchanged:: 3.17.0
1477	        Allow serialization to `float` through use of a new `serialization_type` parameter.
1478	        Defaults to `int` for backwards compatibility. Also affects deserialization.
1479	    .. versionchanged:: 4.0.0
1480	        Remove `serialization_type` parameter and always serialize to float.
1481	        Value is cast to a `float` upon deserialization.
1482	    """
1483	
1484	    WEEKS = "weeks"
1485	    DAYS = "days"
1486	    HOURS = "hours"
1487	    MINUTES = "minutes"
1488	    SECONDS = "seconds"
1489	    MILLISECONDS = "milliseconds"
1490	    MICROSECONDS = "microseconds"
1491	
1492	    # cache this mapping on class level for performance
1493	    _unit_to_microseconds_mapping = {
1494	        WEEKS: 1000000 * 60 * 60 * 24 * 7,
1495	        DAYS: 1000000 * 60 * 60 * 24,
1496	        HOURS: 1000000 * 60 * 60,
1497	        MINUTES: 1000000 * 60,
1498	        SECONDS: 1000000,
1499	        MILLISECONDS: 1000,
1500	        MICROSECONDS: 1,
1501	    }
1502	
1503	    #: Default error messages.
1504	    default_error_messages = {
1505	        "invalid": "Not a valid period of time.",
1506	        "format": "{input!r} cannot be formatted as a timedelta.",
1507	    }
1508	
1509	    def __init__(
1510	        self,
1511	        precision: str = SECONDS,
1512	        **kwargs: Unpack[_BaseFieldKwargs],
1513	    ) -> None:
1514	        precision = precision.lower()
1515	
1516	        if precision not in self._unit_to_microseconds_mapping:
1517	            units = ", ".join(self._unit_to_microseconds_mapping)
1518	            msg = f"The precision must be one of: {units}."
1519	            raise ValueError(msg)
1520	
1521	        self.precision = precision
1522	        super().__init__(**kwargs)
1523	
1524	    def _serialize(self, value, attr, obj, **kwargs) -> float | None:
1525	        if value is None:
1526	            return None
1527	
1528	        # limit float arithmetics to a single division to minimize precision loss
1529	        microseconds: int = utils.timedelta_to_microseconds(value)
1530	        microseconds_per_unit: int = self._unit_to_microseconds_mapping[self.precision]
1531	        return microseconds / microseconds_per_unit
1532	
1533	    def _deserialize(self, value, attr, data, **kwargs) -> dt.timedelta:
1534	        if isinstance(value, dt.timedelta):
1535	            return value
1536	        try:
1537	            value = float(value)
1538	        except (TypeError, ValueError) as error:
1539	            raise self.make_error("invalid") from error

1541	        kwargs = {self.precision: value}
1542	
1543	        try:
1544	            return dt.timedelta(**kwargs)
1545	        except OverflowError as error:

1598	            try:
1599	                self.value_field = _resolve_field_instance(values)
1600	            except _FieldInstanceResolutionError as error:

1604	                ) from error
1605	            if isinstance(self.value_field, Nested):

1607	                self.exclude = self.value_field.exclude
1608	
1609	    def _bind_to_schema(self, field_name, parent):
1610	        super()._bind_to_schema(field_name, parent)
1611	        if self.value_field:
1612	            self.value_field = copy.deepcopy(self.value_field)
1613	            self.value_field._bind_to_schema(field_name, self)
1614	        if isinstance(self.value_field, Nested):
1615	            self.value_field.only = self.only
1616	            self.value_field.exclude = self.exclude
1617	        if self.key_field:
1618	            self.key_field = copy.deepcopy(self.key_field)
1619	            self.key_field._bind_to_schema(field_name, self)

1622	        if value is None:
1623	            return None
1624	        if not self.value_field and not self.key_field:
1625	            return self.mapping_type(value)
1626	
1627	        # Serialize keys
1628	        if self.key_field is None:
1629	            keys = {k: k for k in value}
1630	        else:
1631	            keys = {
1632	                k: self.key_field._serialize(k, None, None, **kwargs) for k in value
1633	            }
1634	
1635	        # Serialize values
1636	        result = self.mapping_type()
1637	        if self.value_field is None:
1638	            for k, v in value.items():
1639	                if k in keys:
1640	                    result[keys[k]] = v
1641	        else:
1642	            for k, v in value.items():
1643	                result[keys[k]] = self.value_field._serialize(v, None, None, **kwargs)
1644	
1645	        return result
1646	
1647	    def _deserialize(self, value, attr, data, **kwargs):
1648	        if not isinstance(value, _Mapping):
1649	            raise self.make_error("invalid")
1650	        if not self.value_field and not self.key_field:
1651	            return self.mapping_type(value)
1652

1660	            for key in value:
1661	                try:
1662	                    keys[key] = self.key_field.deserialize(key, **kwargs)
1663	                except ValidationError as error:
