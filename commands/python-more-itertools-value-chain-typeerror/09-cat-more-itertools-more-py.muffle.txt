321	class peekable:
322	    """Wrap an iterator to allow lookahead and prepending elements.
323	
324	    Call :meth:`peek` on the result to get the value that will be returned
325	    by :func:`next`. This won't advance the iterator:
326	
327	        >>> p = peekable(['a', 'b'])
328	        >>> p.peek()
329	        'a'
330	        >>> next(p)
331	        'a'
332	
333	    Pass :meth:`peek` a default value to return that instead of raising
334	    ``StopIteration`` when the iterator is exhausted.
335	
336	        >>> p = peekable([])
337	        >>> p.peek('hi')
338	        'hi'
339	
340	    peekables also offer a :meth:`prepend` method, which "inserts" items
341	    at the head of the iterable:
342	
343	        >>> p = peekable([1, 2, 3])
344	        >>> p.prepend(10, 11, 12)
345	        >>> next(p)
346	        10
347	        >>> p.peek()
348	        11
349	        >>> list(p)
350	        [11, 12, 1, 2, 3]
351	
352	    peekables can be indexed. Index 0 is the item that will be returned by
353	    :func:`next`, index 1 is the item after that, and so on:
354	    The values up to the given index will be cached.
355	
356	        >>> p = peekable(['a', 'b', 'c', 'd'])
357	        >>> p[0]
358	        'a'
359	        >>> p[1]
360	        'b'
361	        >>> next(p)
362	        'a'
363	
364	    Negative indexes are supported, but be aware that they will cache the
365	    remaining items in the source iterator, which may require significant
366	    storage.
367	
368	    To check whether a peekable is exhausted, check its truth value:
369	
370	        >>> p = peekable(['a', 'b'])
371	        >>> if p:  # peekable has items
372	        ...     list(p)

1134	class bucket:
1135	    """Wrap *iterable* and return an object that buckets the iterable into
1136	    child iterables based on a *key* function.
1137	
1138	        >>> iterable = ['a1', 'b1', 'c1', 'a2', 'b2', 'c2', 'b3']
1139	        >>> s = bucket(iterable, key=lambda x: x[0])  # Bucket by 1st character
1140	        >>> sorted(list(s))  # Get the keys
1141	        ['a', 'b', 'c']
1142	        >>> a_iterable = s['a']
1143	        >>> next(a_iterable)
1144	        'a1'
1145	        >>> next(a_iterable)
1146	        'a2'
1147	        >>> list(s['b'])
1148	        ['b1', 'b2', 'b3']
1149	
1150	    The original iterable will be advanced and its items will be cached until
1151	    they are used by the child iterables. This may require significant storage.
1152	
1153	    By default, attempting to select a bucket to which no items belong  will
1154	    exhaust the iterable and cache all values.
1155	    If you specify a *validator* function, selected buckets will instead be
1156	    checked against it.
1157	
1158	        >>> from itertools import count

1160	        >>> key = lambda x: x % 10  # Bucket by last digit

1171	    dictionary, which is generally easier to work with. If the elements

2251	class numeric_range(Sequence):
2252	    """An extension of the built-in ``range()`` function whose arguments can
2253	    be any orderable numeric type.
2254	
2255	    With only *stop* specified, *start* defaults to ``0`` and *step*
2256	    defaults to ``1``. The output items will match the type of *stop*:
2257	
2258	        >>> list(numeric_range(3.5))
2259	        [0.0, 1.0, 2.0, 3.0]
2260	
2261	    With only *start* and *stop* specified, *step* defaults to ``1``. The
2262	    output items will match the type of *start*:
2263	
2264	        >>> from decimal import Decimal
2265	        >>> start = Decimal('2.1')
2266	        >>> stop = Decimal('5.1')
2267	        >>> list(numeric_range(start, stop))
2268	        [Decimal('2.1'), Decimal('3.1'), Decimal('4.1')]
2269	
2270	    With *start*, *stop*, and *step*  specified the output items will match
2271	    the type of ``start + step``:
2272	
2273	        >>> from fractions import Fraction
2274	        >>> start = Fraction(1, 2)  # Start at 1/2
2275	        >>> stop = Fraction(5, 2)  # End at 5/2
2276	        >>> step = Fraction(1, 2)  # Count by 1/2
2277	        >>> list(numeric_range(start, stop, step))
2278	        [Fraction(1, 2), Fraction(1, 1), Fraction(3, 2), Fraction(2, 1)]
2279	
2280	    If *step* is zero, ``ValueError`` is raised. Negative steps are supported:
2281	
2282	        >>> list(numeric_range(3, -1, -1.0))
2283	        [3.0, 2.0, 1.0, 0.0]
2284	
2285	    Be aware of the limitations of floating-point numbers; the representation
2286	    of the yielded numbers may be surprising.
2287	
2288	    ``datetime.datetime`` objects can be used for *start* and *stop*, if *step*
2289	    is a ``datetime.timedelta`` object:
2290	
2291	        >>> import datetime
2292	        >>> start = datetime.datetime(2019, 1, 1)
2293	        >>> stop = datetime.datetime(2019, 1, 3)
2294	        >>> step = datetime.timedelta(days=1)

2296	        >>> next(items)
2297	        datetime.datetime(2019, 1, 1, 0, 0)
