42	    def __init__(self, parsed: bool = False) -> None:
43	        self._map: dict[Key, int | tuple[int, ...]] = {}
44	        self._body: list[tuple[Key | None, Item]] = []
45	        self._parsed = parsed
46	        self._table_keys: list[Key] = []
47	        # number of already-validated fragments and the temp container they
48	        # were merged into, per out-of-order key; lets parse-time validation
49	        # resume where the previous pass stopped instead of re-merging every
50	        # fragment (quadratic) on each append

61	    def unwrap(self) -> dict[str, Any]:
62	        """Returns as pure python object (ppo)"""
63	        unwrapped: dict[str, Any] = {}
64	        # Resolve each key straight from _map, which already holds the parsed
65	        # Key objects and their body index, instead of via self.items(): the
66	        # inherited MutableMapping iteration goes through __getitem__, which
67	        # rebuilds a SingleKey from the bare string on every key only to throw
68	        # it away. Out-of-order keys (a tuple index) still go through
69	        # OutOfOrderTableProxy so their validation (and fragment merge) runs
70	        # exactly as before. _map iterates in the same insertion order as the
71	        # old self.items().
72	        for key, idx in self._map.items():
73	            if isinstance(idx, tuple):
74	                value: Any = OutOfOrderTableProxy(self, idx)
75	            else:
76	                value = self._body[idx][1]
77	            unwrapped[key.key] = value.unwrap() if hasattr(value, "unwrap") else value

79	        return unwrapped

153	    def _get_last_index_before_table(self) -> int:
154	        last_index = -1
155	        for i, (k, v) in enumerate(self._body):
156	            if isinstance(v, Null):

167	    def _validate_out_of_order_table(self, key: Key | None = None) -> None:
168	        if key is None:
169	            for k in list(self._out_of_order_keys):
170	                assert k is not None
171	                self._validate_out_of_order_table(k)
172	            return
173	        if key not in self._map:
174	            return
175	        current_idx = self._map[key]
176	        if not isinstance(current_idx, tuple):
177	            return
178	        if self._parsed:
179	            # while parsing, every fragment appended to an out-of-order key
180	            # triggers a validation pass; resume from the cached temp
181	            # container so each fragment is merged (and deep-copied) once
182	            # instead of on every later pass. Fragments are only ever
183	            # appended during parsing, so a count prefix stays valid; any
184	            # other mutation clears the cache.
185	            validated, temp = self._validation_cache.get(key, (0, None))
186	            if validated > len(current_idx):
187	                validated, temp = 0, None
188	            try:
189	                temp = OutOfOrderTableProxy.validate(
190	                    self, current_idx[validated:], temp
191	                )
192	            except Exception:
193	                # the temp container may be partially mutated; don't let a
194	                # caught-and-retried failure resume from a poisoned cache
195	                self._validation_cache.pop(key, None)
196	                raise
197	            self._validation_cache[key] = (len(current_idx), temp)
198	            return
199	        OutOfOrderTableProxy.validate(self, current_idx)

397	        for k, v in candidate.value.body:
398	            if k is None:
399	                continue

401	            if k in current.value._map:
402	                existing = current.value.item(k)
403	                if isinstance(existing, (Table, AoT)) != isinstance(v, (Table, AoT)):
404	                    raise KeyAlreadyPresent(k)
405	                if k.is_dotted():
406	                    raise TOMLKitError("Redefinition of an existing table")
407	                if isinstance(existing, Table) and isinstance(v, Table):
408	                    if not existing.is_super_table() and not v.is_super_table():
409	                        # Both sides are concrete `[table]` definitions of the

411	                        raise KeyAlreadyPresent(k)
412	                    # One side is still an implicit/super table, so a duplicate
413	                    # (if any) is nested deeper - keep checking the subtree.
414	                    self._validate_table_candidate(existing, v)

495	        if key is None:

501	        if not isinstance(key, Key):

515	            current_item.trivia.trail += "\n"

518	        for k, v in self._map.items():
519	            if isinstance(v, tuple):

523	                        v_ = v_ + 1

539	    def _insert_at(self, idx: int, key: Key | str, item: Any) -> Container:
540	        if idx > len(self._body) - 1:
541	            raise ValueError(f"Unable to insert at position {idx}")

543	        if not isinstance(key, Key):
544	            key = SingleKey(key)
545	
546	        item = _item(item)
547	
548	        if idx > 0:
549	            previous_item = self._body[idx - 1][1]
550	            if not (
551	                isinstance(previous_item, Whitespace)
552	                or ends_with_whitespace(previous_item)
553	                or isinstance(item, (AoT, Table))
554	                or "\n" in previous_item.trivia.trail
555	            ):
556	                previous_item.trivia.trail += "\n"
557	
558	        # Increment indices after the current index
559	        for k, v in self._map.items():
560	            if isinstance(v, tuple):
561	                new_indices = []
562	                for v_ in v:
563	                    if v_ >= idx:
564	                        v_ = v_ + 1
565	
566	                    new_indices.append(v_)
567	
568	                self._map[k] = tuple(new_indices)
569	            elif v >= idx:

573	            current_idx = self._map[key]
574	            if not isinstance(current_idx, tuple):
575	                current_idx = (current_idx,)

629	                    s += self._render_aot(k, v)
630	                else:
631	                    s += self._render_simple_item(k, v)
632	            else:
633	                s += self._render_simple_item(k, v)
634	
635	        return s

637	    def _render_table(self, key: Key, table: Table, prefix: str | None = None) -> str:
638	        cur = ""
639	
640	        if table.display_name is not None:
641	            _key = table.display_name
642	        else:
643	            _key = key.as_string()
644	
645	            if prefix is not None:
646	                _key = prefix + "." + _key
647	
648	        if (
649	            not table.is_super_table()
650	            or (
651	                any(
652	                    not isinstance(v, (Table, AoT, Whitespace, Null))
653	                    for _, v in table.value.body
654	                )
655	                and not key.is_dotted()
656	            )
657	            or (
658	                any(
659	                    k is not None and k.is_dotted()
660	                    for k, v in table.value.body
661	                    if isinstance(v, Table)
662	                )
663	                and not key.is_dotted()
664	            )
665	        ):
666	            open_, close = "[", "]"
667	            if table.is_aot_element():
668	                open_, close = "[[", "]]"
669	
670	            newline_in_table_trivia = (
671	                "\n" if "\n" not in table.trivia.trail and len(table.value) > 0 else ""
672	            )
673	            cur += (
674	                f"{table.trivia.indent}"
675	                f"{open_}"
676	                f"{decode(_key)}"
677	                f"{close}"
678	                f"{table.trivia.comment_ws}"
679	                f"{decode(table.trivia.comment)}"
680	                f"{table.trivia.trail}"
681	                f"{newline_in_table_trivia}"
682	            )
683	        elif table.trivia.indent == "\n":
684	            cur += table.trivia.indent
685	
686	        for k, v in table.value.body:
687	            if isinstance(v, Table):
688	                if (
689	                    cur.strip(" ")
690	                    and not cur.strip(" ").endswith("\n")
691	                    and "\n" not in v.trivia.indent
692	                ):
693	                    cur += "\n"
694	                assert k is not None
695	                if v.is_super_table():
696	                    if k.is_dotted() and not key.is_dotted():
697	                        # Dotted key inside table
698	                        cur += self._render_table(k, v)
699	                    else:
700	                        cur += self._render_table(k, v, prefix=_key)
701	                else:
702	                    cur += self._render_table(k, v, prefix=_key)
703	            elif isinstance(v, AoT):
704	                if (
705	                    cur.strip(" ")
706	                    and not cur.strip(" ").endswith("\n")
707	                    and "\n" not in v.trivia.indent
708	                ):
709	                    cur += "\n"
710	                assert k is not None
711	                cur += self._render_aot(k, v, prefix=_key)
712	            else:
713	                cur += self._render_simple_item(
714	                    k, v, prefix=_key if key.is_dotted() else None
715	                )
716	
717	        return cur

799	    def __contains__(self, key: object) -> bool:
800	        # Native membership test. The inherited ``MutableMapping.__contains__``
801	        # resolves the value via ``__getitem__``/``item()`` (and builds a
802	        # ``NonExistentKey`` on every absent key) only to discard it. Resolve the
803	        # key the same way ``item()`` does -- ``str`` becomes a ``SingleKey``
804	        # (a non-str/non-``Key`` argument still raises ``TypeError``) -- then
805	        # probe ``_map`` directly. For an out-of-order table the proxy is still
806	        # built so its validation runs exactly as before.
807	        if not isinstance(key, Key):
808	            key = SingleKey(key)  # type: ignore[arg-type]
809	        idx = self._map.get(key)
810	        if idx is None:
811	            return False
812	        if isinstance(idx, tuple):
813	            # during parsing every fragment append re-probes membership;
814	            # if the validation cache covers exactly these fragments they
815	            # are already known to merge cleanly, so skip rebuilding the
816	            # proxy (which is linear in the number of fragments)
817	            validated, _ = self._validation_cache.get(key, (0, None))
818	            if not (self._parsed and validated == len(idx)):
819	                OutOfOrderTableProxy(self, idx)
820	        return True

838	        if not isinstance(key, Key):
839	            key = SingleKey(key)

847	    def _replace_at(
848	        self, idx: int | tuple[int, ...], new_key: Key | str, value: Item
849	    ) -> None:
850	        value = _item(value)
851	        self._validation_cache.clear()
852	
853	        if isinstance(idx, tuple):
854	            for i in idx[1:]:
855	                self._body[i] = (None, Null())
856	
857	            idx = idx[0]
858	
859	        k, v = self._body[idx]
860	        assert k is not None
861	        # A dotted key renders its value inline (e.g. ``a.b = 1``), which is only
862	        # consistent with a super table. When the replacement value renders with
863	        # its own ``[header]`` instead (a non-super table), keeping the dotted key
864	        # duplicates the prefix onto the header (#524). Drop the dotted key so the
865	        # replacement renders as a plain table.
866	        dotted_to_header = (
867	            k.is_dotted() and isinstance(value, Table) and not value.is_super_table()
868	        )
869	        # That new header also captures every sibling that renders inline -- plain
870	        # values and dotted keys -- if any still follow it (#513), so it must be
871	        # moved past them, exactly as a value-to-table change already is.
872	        reposition_dotted = dotted_to_header and any(
873	            not isinstance(cur_val, (Null, Whitespace))
874	            and not (
875	                isinstance(cur_val, (Table, AoT))
876	                and (cur_key is None or not cur_key.is_dotted())
877	            )
878	            for cur_key, cur_val in self._body[idx + 1 :]
879	        )
880	        if not isinstance(new_key, Key):
881	            if (
882	                isinstance(value, (AoT, Table)) != isinstance(v, (AoT, Table))
883	                or new_key != k.key
884	                or dotted_to_header
885	            ):
886	                new_key = SingleKey(new_key)
887	            else:  # Inherit the sep of the old key
888	                new_key = k
889	
890	        del self._map[k]
891	        self._map[new_key] = idx
892	        if new_key != k:
893	            dict.__delitem__(self, k.key)
894	
895	        if (
896	            isinstance(value, (AoT, Table)) != isinstance(v, (AoT, Table))
897	            or reposition_dotted
898	        ):
899	            self.remove(k)
900	            if isinstance(value, (AoT, Table)):
901	                # New tables must appear after all entries that render inline:
902	                # plain values and dotted keys (which are super tables). Skip
903	                # those and insert before the first real ``[header]`` so the new
904	                # table cannot swallow a following sibling on round-trip.
905	                for i in range(idx, len(self._body)):
906	                    cur_key, cur_val = self._body[i]
907	                    if isinstance(cur_val, (AoT, Table)) and not (
908	                        cur_key is not None and cur_key.is_dotted()
909	                    ):
910	                        self._insert_at(i, new_key, value)
911	                        idx = i
912	                        break
913	                else:
914	                    idx = -1
915	                    self.append(new_key, value)
916	            else:
917	                # the replaced table's slot lies in the table region, where a
918	                # plain value would be captured by the preceding table on
919	                # round-trip; append() puts it with the other root-level values
920	                idx = -1
921	                self.append(new_key, value)
922	        else:
923	            # Copying trivia
924	            if not isinstance(value, (Whitespace, AoT)):
925	                value.trivia.indent = v.trivia.indent
926	                value.trivia.comment_ws = value.trivia.comment_ws or v.trivia.comment_ws
927	                value.trivia.comment = value.trivia.comment or v.trivia.comment
928	                value.trivia.trail = v.trivia.trail
929	            self._body[idx] = (new_key, value)
930	
931	        if value is not v and hasattr(value, "invalidate_display_name"):
932	            value.invalidate_display_name()
933	
934	        if isinstance(value, Table):
935	            # Insert a cosmetic new line for tables if:
936	            # - it does not have it yet OR is not followed by one
937	            # - it is not the last item, or
938	            # - The table being replaced has a newline
939	            result = self._previous_item_with_index()
940	            assert result is not None
941	            last, _ = result
942	            idx = last if idx < 0 else idx
943	            has_ws = ends_with_whitespace(value)
944	            replace_has_ws = (
945	                isinstance(v, Table)
946	                and v.value.body
947	                and isinstance(v.value.body[-1][1], Whitespace)
948	            )
949	            next_ws = idx < last and isinstance(self._body[idx + 1][1], Whitespace)
950	            if (idx < last or replace_has_ws) and not (next_ws or has_ws):
951	                value.append(None, Whitespace("\n"))
952	
953	            assert isinstance(new_key, Key)
954	            dict.__setitem__(self, new_key.key, value.value)

1030	class OutOfOrderTableProxy(_CustomDict):  # type: ignore[type-arg]
1031	    @staticmethod
1032	    def validate(
1033	        container: Container,
1034	        indices: tuple[int, ...],
1035	        temp_container: Container | None = None,
1036	    ) -> Container:
1037	        """Validate out of order tables in the given container"""
1038	        # Append all items to a temp container to see if there is any error.
1039	        # We deep-copy each value before appending: appending a super table
1040	        # merges it in place into a matching one already in temp_container, and
1041	        # those values are the live subtables of `container`. Without the copy
1042	        # the merge would mutate the caller's tables (and corrupt a later
1043	        # validation pass). The container merge itself is now copy-free for
1044	        # speed, so this is where the isolation lives.
1045	        # Passing a previous pass's temp container back in (with the already
1046	        # validated indices stripped from `indices`) resumes the validation
1047	        # incrementally.
1048	        if temp_container is None:
1049	            temp_container = Container(True)
1050	        for i in indices:
1051	            _, item = container._body[i]
1052	
1053	            if isinstance(item, Table):
1054	                for k, v in item.value.body:
1055	                    temp_container.append(k, copy.deepcopy(v), validate=True)
1056	
1057	        temp_container._validate_out_of_order_table()
1058	        return temp_container
1059	
1060	    def __init__(self, container: Container, indices: tuple[int, ...]) -> None:
1061	        self._container = container
1062	        self._internal_container = Container(True)
1063	        self._tables: list[Table] = []
1064	        self._tables_map: dict[Key, list[int]] = {}
1065	
1066	        for i in indices:
1067	            _, _item = self._container._body[i]
1068	
1069	            if isinstance(_item, Table):
1070	                self._tables.append(_item)
1071	                table_idx = len(self._tables) - 1
1072	                for k, v in _item.value.body:
1073	                    merged = self._merge_aot_fragment(k, v)
1074	                    if merged is None:
1075	                        self._internal_container._raw_append(k, v)
1076	                    else:
1077	                        v = merged
1078	                    key_indices = self._tables_map.setdefault(k, [])  # type: ignore[arg-type]
1079	                    if table_idx not in key_indices:
1080	                        key_indices.append(table_idx)
1081	                    if k is not None:
1082	                        dict.__setitem__(self, k.key, v)
1083	
1084	        self._internal_container._validate_out_of_order_table()
1085	
1086	    def _merge_aot_fragment(self, key: Key | None, item: Item) -> AoT | None:
1087	        """
1088	        Merge an array-of-tables fragment from a later out-of-order table part.
1089	
1090	        An AoT whose elements are split across out-of-order parts of the same
1091	        table arrives here once per part; ``_raw_append`` only knows how to
1092	        chain duplicate ``Table`` parts and would raise ``KeyAlreadyPresent``.
1093	        The fragments are presented as a new merged ``AoT`` referencing the
1094	        live element tables, without mutating either fragment (the parts keep
1095	        rendering their own elements).
1096	
1097	        Returns the merged ``AoT``, or ``None`` if this is not such a fragment.
1098	        """
1099	        internal = self._internal_container
1100	        if key is None or not isinstance(item, AoT) or key not in internal._map:
1101	            return None
1102	        idx = internal._map[key]
1103	        if isinstance(idx, tuple):
1104	            # A tuple index means the key already resolved to several body
1105	            # positions, which here only happens for a degenerate collision
1106	            # (e.g. a non-AoT part sharing the key). Three or more genuine AoT
1107	            # fragments do not reach this branch: each later fragment merges
1108	            # into the single growing AoT below, so ``idx`` stays an int.
1109	            return None
1110	        existing = internal._body[idx][1]
1111	        if not isinstance(existing, AoT):
1112	            return None
1113	
1114	        merged = AoT([*existing.body, *item.body], parsed=True)
1115	        internal._body[idx] = (internal._body[idx][0], merged)
1116	        dict.__setitem__(internal, key.key, merged.value)
1117	        return merged

1147	    def __setitem__(self, key: Key | str, value: Any) -> None:
1148	        from .items import item as _item_fn
1149	
1150	        def _is_table_or_aot(it: Any) -> bool:
1151	            return isinstance(_item_fn(it), (Table, AoT))
1152	
1153	        _key: Key = key if isinstance(key, Key) else SingleKey(key)
1154	
1155	        if _key in self._tables_map:
1156	            # Overwrite the first table and remove others
1157	            map_indices = self._tables_map[_key]
1158	            while len(map_indices) > 1:
1159	                table = self._tables[map_indices.pop()]
1160	                self._remove_table(table)
1161	            old_value = self._tables[map_indices[0]][key]
1162	            if _is_table_or_aot(old_value) and not _is_table_or_aot(value):
1163	                # Remove the entry from the map and set value again.
1164	                del self._tables[map_indices[0]][key]
1165	                del self._tables_map[_key]
1166	                self[key] = value
1167	                return
1168	            self._tables[map_indices[0]][key] = value
1169	        elif self._tables:
1170	            if not _is_table_or_aot(value):  # if the value is a plain value
1171	                for table in self._tables:
1172	                    # find the first table that allows plain values
1173	                    if any(not _is_table_or_aot(v) for _, v in table.items()):
1174	                        table[key] = value
1175	                        break
1176	                else:
1177	                    # No part holds a plain value yet, so the chosen part must
1178	                    # start rendering its own ``[key]`` header. Prefer a part
1179	                    # that is not a super table (it already renders that header):
1180	                    # turning a header-less super part concrete here would emit a
1181	                    # second, duplicate header next to an existing concrete part
1182	                    # and produce TOML that no longer parses.
1183	                    for table in self._tables:
1184	                        if not table.is_super_table():
1185	                            table[key] = value
1186	                            break
1187	                    else:
1188	                        self._tables[0][key] = value
1189	            else:
1190	                self._tables[0][key] = value
1191	        else:
1192	            self._container[key] = value
1193	
1194	        self._internal_container[key] = value
1195	        if key is not None:
1196	            dict.__setitem__(self, key, value)

1198	    def _remove_table(self, table: Table) -> None:
1199	        """Remove table from the parent container"""
1200	        self._tables.remove(table)
1201	        for idx, body_item in enumerate(self._container._body):
