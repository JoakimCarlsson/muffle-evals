5	/**
6	 * Parse options.
7	 */
8	export interface CookieParseOptions {
9	  /**
10	   * Specifies a function that will be used to decode a [cookie-value](https://datatracker.ietf.org/doc/html/rfc6265#section-4.1.1).
11	   * Since the value of a cookie has a limited character set (and must be a simple string), this function can be used to decode
12	   * a previously-encoded cookie value into a JavaScript string.
13	   *
14	   * The default function is the global `decodeURIComponent`, wrapped in a `try..catch`. If an error
15	   * is thrown it will return the cookie's original value. If you provide your own encode/decode
16	   * scheme you must ensure errors are appropriately handled.
17	   *
18	   * @default decode
19	   */
20	  decode?: (str: string) => string | undefined;
21	
22	  /**
23	   * Custom function to filter parsing specific keys.
24	   */
25	  filter?(key: string): boolean;
26	}

47	/**
48	 * Set-Cookie object.
49	 */
50	export interface SetCookie {
51	  /**
52	   * Specifies the name of the cookie.
53	   */
54	  name: string;
55	  /**
56	   * Specifies the string to be the value for the cookie.
57	   */
58	  value: string | undefined;
59	  /**
60	   * Specifies the `number` (in seconds) to be the value for the [`Max-Age` `Set-Cookie` attribute](https://tools.ietf.org/html/rfc6265#section-5.2.2).
61	   *
62	   * The [cookie storage model specification](https://tools.ietf.org/html/rfc6265#section-5.3) states that if both `expires` and
63	   * `maxAge` are set, then `maxAge` takes precedence, but it is possible not all clients by obey this,
64	   * so if both are set, they should point to the same date and time.
65	   */
66	  maxAge?: number;
67	  /**
68	   * Specifies the `Date` object to be the value for the [`Expires` `Set-Cookie` attribute](https://tools.ietf.org/html/rfc6265#section-5.2.1).
69	   * When no expiration is set, clients consider this a "non-persistent cookie" and delete it when the current session is over.
70	   *
71	   * The [cookie storage model specification](https://tools.ietf.org/html/rfc6265#section-5.3) states that if both `expires` and
72	   * `maxAge` are set, then `maxAge` takes precedence, but it is possible not all clients by obey this,
73	   * so if both are set, they should point to the same date and time.
74	   */
75	  expires?: Date;
76	  /**
77	   * Specifies the value for the [`Domain` `Set-Cookie` attribute](https://tools.ietf.org/html/rfc6265#section-5.2.3).
78	   * When no domain is set, clients consider the cookie to apply to the current domain only.
79	   */
80	  domain?: string;
81	  /**
82	   * Specifies the value for the [`Path` `Set-Cookie` attribute](https://tools.ietf.org/html/rfc6265#section-5.2.4).
83	   * When no path is set, the path is considered the ["default path"](https://tools.ietf.org/html/rfc6265#section-5.1.4).
84	   */
85	  path?: string;
86	  /**
87	   * Enables the [`HttpOnly` `Set-Cookie` attribute](https://tools.ietf.org/html/rfc6265#section-5.2.6).
88	   * When enabled, clients will not allow client-side JavaScript to see the cookie in `document.cookie`.
89	   */
90	  httpOnly?: boolean;
91	  /**
92	   * Enables the [`Secure` `Set-Cookie` attribute](https://tools.ietf.org/html/rfc6265#section-5.2.5).
93	   * When enabled, clients will only send the cookie back if the browser has an HTTPS connection.
94	   */
95	  secure?: boolean;
96	  /**
97	   * Enables the [`Partitioned` `Set-Cookie` attribute](https://tools.ietf.org/html/draft-cutler-httpbis-partitioned-cookies/).
98	   * When enabled, clients will only send the cookie back when the current domain _and_ top-level domain matches.
99	   *
100	   * This is an attribute that has not yet been fully standardized, and may change in the future.
101	   * This also means clients may ignore this attribute until they understand it. More information
102	   * about can be found in [the proposal](https://github.com/privacycg/CHIPS).
103	   */
104	  partitioned?: boolean;
105	  /**
106	   * Specifies the value for the [`Priority` `Set-Cookie` attribute](https://tools.ietf.org/html/draft-west-cookie-priority-00#section-4.1).
107	   *
108	   * - `'low'` will set the `Priority` attribute to `Low`.
109	   * - `'medium'` will set the `Priority` attribute to `Medium`, the default priority when not set.
110	   * - `'high'` will set the `Priority` attribute to `High`.
111	   *
112	   * More information about priority levels can be found in [the specification](https://tools.ietf.org/html/draft-west-cookie-priority-00#section-4.1).
113	   */
114	  priority?: "low" | "medium" | "high";
115	  /**
116	   * Specifies the value for the [`SameSite` `Set-Cookie` attribute](https://tools.ietf.org/html/draft-ietf-httpbis-rfc6265bis-09#section-5.4.7).
117	   *
118	   * - `true` will set the `SameSite` attribute to `Strict` for strict same site enforcement.
119	   * - `'lax'` will set the `SameSite` attribute to `Lax` for lax same site enforcement.
120	   * - `'none'` will set the `SameSite` attribute to `None` for an explicit cross-site cookie.
121	   * - `'strict'` will set the `SameSite` attribute to `Strict` for strict same site enforcement.
122	   *
123	   * More information about enforcement levels can be found in [the specification](https://tools.ietf.org/html/draft-ietf-httpbis-rfc6265bis-09#section-5.4.7).
124	   */
125	  sameSite?: boolean | "lax" | "strict" | "none";
126	}
