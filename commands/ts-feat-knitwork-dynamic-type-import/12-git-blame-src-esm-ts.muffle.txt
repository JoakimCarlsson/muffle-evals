26	/**
27	 * Generate an ESM `import` statement.
28	 *
29	 * @example
30	 *
31	 * ```js
32	 * genImport("pkg", "foo");
33	 * // ~> `import foo from "pkg";`
34	 *
35	 * genImport("pkg", ["foo"]);
36	 * // ~> `import { foo } from "pkg";`
37	 *
38	 * genImport("pkg", ["a", "b"]);
39	 * // ~> `import { a, b } from "pkg`;
40	 *
41	 * genImport("pkg", [{ name: "default", as: "bar" }]);
42	 * // ~> `import { default as bar } from "pkg`;
43	 *
44	 * genImport("pkg", [{ name: "foo", as: "bar" }]);
45	 * // ~> `import { foo as bar } from "pkg`;
46	 *
47	 * genImport("pkg", "foo", { attributes: { type: "json" } });
48	 * // ~> `import foo from "pkg" with { type: "json" };
49	 *
50	 * genExport("pkg", "foo");
51	 * // ~> `export foo from "pkg";`
52	 *
53	 * genExport("pkg", ["a", "b"]);
54	 * // ~> `export { a, b } from "pkg";`
55	 *
56	 * // export * as bar from "pkg"
57	 * genExport("pkg", { name: "*", as: "bar" });
58	 * // ~> `export * as bar from "pkg";`
59	 * ```
60	 *
61	 * @group ESM
62	 */
63	export function genImport(

97	/**
98	 * Generate an ESM dynamic `import()` statement.
99	 *
100	 * @group ESM
101	 */
102	export function genDynamicImport(
103	  specifier: string,
104	  options: DynamicImportOptions = {},
105	) {
106	  const commentString = options.comment ? ` /* ${options.comment} */` : "";
107	  const wrapperString = options.wrapper === false ? "" : "() => ";
108	  const ineropString = options.interopDefault
109	    ? ".then(m => m.default || m)"
110	    : "";
111	  const optionsString = _genDynamicImportAttributes(options);
112	  return `${wrapperString}import(${genString(
113	    specifier,
114	    options,
115	  )}${commentString}${optionsString})${ineropString}`;
116	}
