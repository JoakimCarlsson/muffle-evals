11	 * Normalises alias mappings, ensuring that more specific aliases are resolved before less specific ones.
12	 * This function also ensures that aliases do not resolve to themselves cyclically.
13	 *
14	 * @param _aliases - A set of alias mappings where each key is an alias and its value is the actual path it points to.
15	 * @returns a set of normalised alias mappings.
16	 */
17	export function normalizeAliases(_aliases: Record<string, string>) {
18	  if ((_aliases as any)[normalizedAliasSymbol]) {
19	    return _aliases;
20	  }
21	
22	  // Sort aliases from specific to general (ie. fs/promises before fs)
23	  const aliases = Object.fromEntries(
24	    Object.entries(_aliases).sort(([a], [b]) => _compareAliases(a, b)),
25	  );
26	
27	  // Resolve alias values in relation to each other
28	  for (const key in aliases) {
29	    for (const alias in aliases) {
30	      // don't resolve a more specific alias with regard to a less specific one
31	      if (alias === key || key.startsWith(alias)) {
32	        continue;
33	      }
34	
35	      if (
36	        aliases[key].startsWith(alias) &&
37	        pathSeparators.has(aliases[key][alias.length])
38	      ) {
39	        aliases[key] = aliases[alias] + aliases[key].slice(alias.length);
40	      }
41	    }
42	  }
43	
44	  Object.defineProperty(aliases, normalizedAliasSymbol, {
45	    value: true,
46	    enumerable: false,
47	  });
