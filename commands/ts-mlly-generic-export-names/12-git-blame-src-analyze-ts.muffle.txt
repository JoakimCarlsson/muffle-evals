389	/**
390	 * Extracts extra variable names from a comma-separated declaration list,
391	 * ignoring commas nested inside parentheses, brackets, braces, generics, or strings.
392	 *
393	 * For `export const foo = fn('a', bar), baz = 2`, only `baz` should be
394	 * extracted — commas inside `fn(...)` or `Handler<A, B>` are not declaration separators.
395	 * @param {string} extraNamesStr - The string containing the extra variable declarations to parse.
396	 * @returns {string[]} An array of variable names extracted from the declaration list.
397	 */
398	function _extractExtraNames(extraNamesStr: string): string[] {
399	  const names: string[] = [];
400	  let depth = 0;
401	  let angleDepth = 0;
402	  let inString: string | false = false;
403	  let inTypeAnnotation = false;
404	
405	  for (let i = 0; i < extraNamesStr.length; i++) {
406	    const char = extraNamesStr[i];
407	
408	    // Handle string literals — skip their contents entirely
409	    if (inString) {
410	      if (char === inString) {
411	        let backslashCount = 0;
412	        for (let j = i - 1; j >= 0 && extraNamesStr[j] === "\\"; j--) {
413	          backslashCount++;
414	        }
415	        if (backslashCount % 2 === 0) {
416	          inString = false;
417	        }
418	      }
419	      continue;
420	    }
421	    if (char === '"' || char === "'" || char === "`") {
422	      inString = char;
423	      continue;
424	    }
425	
426	    // Track type annotations (between `:` and `=` at depth 0) for generic angle brackets
427	    if (char === ":" && depth === 0 && angleDepth === 0) {
428	      inTypeAnnotation = true;
429	      continue;
430	    }
431	    if (
432	      char === "=" &&
433	      extraNamesStr[i + 1] !== ">" &&
434	      depth === 0 &&
435	      angleDepth === 0
436	    ) {
437	      inTypeAnnotation = false;
438	    }
439	    if (inTypeAnnotation && char === "<") {
440	      angleDepth++;
441	      continue;
442	    }
443	    if (inTypeAnnotation && char === ">" && angleDepth > 0) {
444	      angleDepth--;
445	      continue;
446	    }
447	
448	    // Track bracket nesting
449	    if (char === "(" || char === "[" || char === "{") {
450	      depth++;
451	    } else if (char === ")" || char === "]" || char === "}") {
452	      depth--;
453	    }
454	
455	    // A comma at depth 0 (outside generics) is a real declaration separator
456	    if (char === "," && depth === 0 && angleDepth === 0) {
457	      // Extract the identifier that follows this comma
458	      const rest = extraNamesStr.slice(i + 1);
459	      const match = rest.match(/^\s*([\w$]+)/);
460	      if (match) {
461	        names.push(match[1]);
462	      }
463	    }
464	  }
465	
466	  return names;
467	}

469	/**
470	 * Identifies all export statements in the supplied source code and categorises them into different types such as declarations, named, default and star exports.
471	 * This function processes the code to capture different forms of export statements and normalise their representation for further processing.
472	 *
473	 * @param {string} code - The source code containing the export statements to be analysed.
474	 * @returns {ESMExport[]} An array of {@link ESMExport} objects representing each export found, properly categorised and structured.
475	 */
476	export function findExports(code: string): ESMExport[] {
477	  // Find declarations like export const foo = 'bar'
478	  const declaredExports: DeclarationExport[] = matchAll(EXPORT_DECAL_RE, code, {
479	    type: "declaration",
480	  });
481	  // Parse extra names (foo, bar)
482	  for (const declaredExport of declaredExports) {
483	    // function declarations don't have extra names
484	    if (/^export\s+(?:async\s+)?function/.test(declaredExport.code)) {
485	      continue;
486	    }
487	    const extraNamesStr = (declaredExport as any).extraNames as
488	      | string
489	      | undefined;
490	    if (extraNamesStr) {
491	      const extraNames = _extractExtraNames(extraNamesStr);
492	      if (extraNames.length > 0) {
493	        declaredExport.names = [declaredExport.name, ...extraNames];
494	      }
495	    }
496	    delete (declaredExport as any).extraNames;
497	  }
498	
499	  // Find named exports
500	  const namedExports: NamedExport[] = normalizeNamedExports(
501	    matchAll(EXPORT_NAMED_RE, code, {
502	      type: "named",
503	    }),
504	  );
505	
506	  const destructuredExports: NamedExport[] = matchAll(
507	    EXPORT_NAMED_DESTRUCT,
508	    code,
509	    { type: "named" },
510	  );
511	  for (const namedExport of destructuredExports) {
512	    // @ts-expect-error groups
513	    namedExport.exports = namedExport.exports1 || namedExport.exports2;
514	    namedExport.names = namedExport.exports
515	      .replace(/^\r?\n?/, "")
516	      .split(/\s*,\s*/g)
517	      .filter((name) => !TYPE_RE.test(name))
518	      .map((name) =>
519	        name
520	          .replace(/^.*?\s*:\s*/, "")
521	          .replace(/\s*=\s*.*$/, "")
522	          .trim(),
523	      );
524	  }
525	
526	  // Find export default
527	  const defaultExport: DefaultExport[] = matchAll(EXPORT_DEFAULT_RE, code, {
528	    type: "default",
529	    name: "default",
530	  });
531	
532	  // Find export default class
533	  const defaultClassExports = matchAll(EXPORT_DEFAULT_CLASS_RE, code, {
534	    type: "declaration",
535	  });
536	
537	  // Find export star
538	  const starExports: ESMExport[] = matchAll(EXPORT_STAR_RE, code, {
539	    type: "star",
540	  });
541	
542	  // Merge and normalize exports
543	
544	  const exports: ESMExport[] = normalizeExports([
545	    ...declaredExports,
546	    ...namedExports,
547	    ...destructuredExports,
548	    ...defaultExport,
549	    ...defaultClassExports,
550	    ...starExports,
551	  ]);
552	
553	  // Return early when there is no  export statement
554	  if (exports.length === 0) {
555	    return [];
556	  }
557	  const exportLocations = _tryGetLocations(code, "export");
558	  if (exportLocations && exportLocations.length === 0) {
559	    return [];
560	  }
561	
562	  return (
563	    // Filter false positive export matches
564	    _filterStatement(exportLocations, exports)
565	      // Prevent multiple exports of same function, only keep latest iteration of signatures
566	      .filter((exp, index, exports) => {
567	        const nextExport = exports[index + 1];
568	        return (
569	          !nextExport ||
570	          exp.type !== nextExport.type ||
571	          !exp.name ||
572	          exp.name !== nextExport.name
573	        );
574	      })
575	  );
576	}

631	function normalizeExports(exports: (ESMExport & { declaration?: string })[]) {
632	  for (const exp of exports) {
633	    if (!exp.name && exp.names && exp.names.length === 1) {
634	      exp.name = exp.names[0];
635	    }
636	    if (exp.name === "default" && exp.type !== "default") {
637	      exp._type = exp.type;
638	      exp.type = "default";
639	    }
640	    if (!exp.names && exp.name) {
641	      exp.names = [exp.name];
642	    }
643	    if (exp.type === "declaration" && exp.declaration) {
644	      exp.declarationType = exp.declaration.replace(
645	        /^declare\s*/,
646	        "",
647	      ) as ESMExport["declarationType"];

678	 * Asynchronously resolves and returns all export names from a module specified by its module identifier.
679	 * This function recursively resolves all explicitly named and asterisked (* as) exports to fully enumerate the exported identifiers.
680	 *
681	 * @param {string} id - The module identifier to resolve.
682	 * @param {ResolveOptions} [options] - Optional settings for resolving the module path, such as the base URL.
683	 * @returns {Promise<string[]>} A promise that resolves to an array of export names from the module.
684	 */
685	export async function resolveModuleExportNames(
686	  id: string,
687	  options?: ResolveOptions,
688	): Promise<string[]> {
689	  const url = await resolvePath(id, options);
690	  const code = await loadURL(url);
691	  const exports = findExports(code);
692	
693	  // Explicit named exports
694	  const exportNames = new Set(
695	    exports.flatMap((exp) => exp.names).filter(Boolean),
696	  );
697	
698	  // Recursive * exports
699	  for (const exp of exports) {
700	    if (exp.type !== "star" || !exp.specifier) {
701	      continue;
702	    }
703	    const subExports = await resolveModuleExportNames(exp.specifier, {
704	      ...options,
705	      url,

708	      exportNames.add(subExport);
709	    }
710	  }

712	  return [...exportNames];
