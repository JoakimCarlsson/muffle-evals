48	/**
49	 * Checks if the input has a protocol.
50	 *
51	 * You can use `{ acceptRelative: true }` to accept relative URLs as valid protocol.
52	 *
53	 * @example
54	 *
55	 * ```js
56	 * hasProtocol('https://example.com'); // true
57	 *
58	 * hasProtocol("//example.com"); // false
59	 *
60	 * hasProtocol('//example.com', { acceptRelative: true });  // true
61	 *
62	 * hasProtocol("ftp://example.com"); // true
63	 *
64	 * hasProtocol('data:text/plain'); // true
65	 *
66	 * hasProtocol('data:text/plain', { strict: true }); // false
67	 * ```
68	 *
69	 * @group utils
70	 */
71	export function hasProtocol(
72	  inputString: string,
73	  opts: boolean | HasProtocolOptions = {},
74	): boolean {
75	  if (typeof opts === "boolean") {
76	    opts = { acceptRelative: opts };
77	  }
78	  if (opts.strict) {
79	    return PROTOCOL_STRICT_REGEX.test(inputString);
80	  }
81	  return (
82	    PROTOCOL_REGEX.test(inputString) ||
83	    (opts.acceptRelative ? PROTOCOL_RELATIVE_REGEX.test(inputString) : false)
84	  );
85	}

137	/**
138	 * Removes the trailing slash from the URL or pathname.
139	 *
140	 * If second argument is `true`, it will only remove the trailing slash if it's not part of the query or fragment with cost of more expensive operations.
141	 *
142	 * @example
143	 *
144	 * ```js
145	 * withoutTrailingSlash("/foo/"); // "/foo"
146	 *
147	 * withoutTrailingSlash("/path/?query=true", true); // "/path?query=true"
148	 * ```
149	 *
150	 * @group utils
151	 */
152	export function withoutTrailingSlash(
153	  input = "",
154	  respectQueryAndFragment?: boolean,
155	): string {
156	  if (!respectQueryAndFragment) {
157	    return (hasTrailingSlash(input) ? input.slice(0, -1) : input) || "/";
158	  }
159	  if (!hasTrailingSlash(input, true)) {
160	    return input || "/";
161	  }
162	  let path = input;
163	  let fragment = "";
164	  const fragmentIndex = input.indexOf("#");
165	  if (fragmentIndex !== -1) {
166	    path = input.slice(0, fragmentIndex);
167	    fragment = input.slice(fragmentIndex);
168	  }
169	  const [s0, ...s] = path.split("?");
170	  const cleanPath = s0.endsWith("/") ? s0.slice(0, -1) : s0;

172	    (cleanPath || "/") + (s.length > 0 ? `?${s.join("?")}` : "") + fragment

178	 *
179	 * If second argument is `true`, it will only add the trailing slash if it's not part of the query or fragment with cost of more expensive operation.

210	  }
211	  const [s0, ...s] = path.split("?");
212	  return s0 + "/" + (s.length > 0 ? `?${s.join("?")}` : "") + fragment;

434	/**
435	 * Joins multiple URL segments into a single URL and also handles relative paths with `./` and `../`.
436	 *
437	 * @example
438	 *
439	 * ```js
440	 * joinRelativeURL("/a", "../b", "./c"); // "/b/c"
441	 * ```
442	 *
443	 * @group utils
444	 */
445	export function joinRelativeURL(..._input: string[]): string {
446	  // Inlined regex to increase browser compatibility
447	  const JOIN_SEGMENT_SPLIT_RE = /\/(?!\/)/;
448	
449	  const input = _input.filter(Boolean);
450	
451	  const segments: string[] = [];
452	
453	  let segmentsDepth = 0;
454	
455	  for (const i of input) {
456	    if (!i || i === "/") {
457	      continue;
458	    }
459	    for (const [sindex, s] of i.split(JOIN_SEGMENT_SPLIT_RE).entries()) {
460	      if (!s || s === ".") {
461	        continue;
462	      }
463	      if (s === "..") {
464	        if (segments.length === 1 && hasProtocol(segments[0])) {
465	          continue;
466	        }
467	        segments.pop();
468	        segmentsDepth--;
469	        continue;
470	      }
471	      // eslint-disable-next-line unicorn/prefer-at
472	      if (sindex === 1 && segments[segments.length - 1]?.endsWith(":/")) {
473	        segments[segments.length - 1] += "/" + s;
474	        continue;
475	      }
476	      segments.push(s);
477	      segmentsDepth++;
478	    }
479	  }
480	
481	  let url = segments.join("/");
482	
483	  if (segmentsDepth >= 0) {
484	    // Preserve leading slash
485	    if (input[0]?.startsWith("/") && !url.startsWith("/")) {
486	      url = "/" + url;
487	    } else if (input[0]?.startsWith("./") && !url.startsWith("./")) {
488	      url = "./" + url;
489	    }
490	  } else {
491	    // Add relative prefix
492	    url = "../".repeat(-1 * segmentsDepth) + url;
493	  }
494	
495	  // Preserve trailing slash
496	  // eslint-disable-next-line unicorn/prefer-at
497	  if (input[input.length - 1]?.endsWith("/") && !url.endsWith("/")) {
498	    url += "/";
499	  }
500	
501	  return url;
502	}

595	/**
596	 * Resolves multiple URL segments into a single URL.
597	 *
598	 * @example
599	 *
600	 * ```js
601	 * resolveURL("http://foo.com/foo?test=123#token", "bar", "baz");
602	 * // Returns "http://foo.com/foo/bar/baz?test=123#token"
603	 * ```
604	 *
605	 * @group utils
606	 */
607	export function resolveURL(base = "", ...inputs: string[]): string {
608	  if (typeof base !== "string") {
609	    throw new TypeError(
610	      `URL input should be string received ${typeof base} (${base})`,
611	    );
612	  }
613	
614	  const filteredInputs = inputs.filter((input) => isNonEmptyURL(input));
615	
616	  if (filteredInputs.length === 0) {
617	    return base;
618	  }
619	
620	  const url = parseURL(base);
621	
622	  for (const inputSegment of filteredInputs) {
623	    const urlSegment = parseURL(inputSegment);
624	
625	    // Append path
626	    if (urlSegment.pathname) {
627	      url.pathname =
628	        withTrailingSlash(url.pathname) +
629	        withoutLeadingSlash(urlSegment.pathname);
630	    }
631	
632	    // Override hash
633	    if (urlSegment.hash && urlSegment.hash !== "#") {
634	      url.hash = urlSegment.hash;
635	    }
636	
637	    // Append search
638	    if (urlSegment.search && urlSegment.search !== "?") {
639	      if (url.search && url.search !== "?") {
640	        const queryString = stringifyQuery({
641	          ...parseQuery(url.search),
642	          ...parseQuery(urlSegment.search),
643	        });
644	        url.search = queryString.length > 0 ? "?" + queryString : "";
645	      } else {
646	        url.search = urlSegment.search;
647	      }
648	    }
649	  }
650	
651	  return stringifyParsedURL(url);
652	}

670	  leadingSlash?: boolean;

674	/**
675	 *  Checks if two paths are equal regardless of encoding, trailing slash, and leading slash differences.
676	 *
677	 * You can make slash check strict by setting `{ trailingSlash: true, leadingSlash: true }` as options.
678	 *
679	 * You can make encoding check strict by setting `{ encoding: true }` as options.
680	 *
681	 * @example
682	 *
683	 * ```js
684	 * isEqual("/foo", "foo"); // true
685	 * isEqual("foo/", "foo"); // true
686	 * isEqual("/foo bar", "/foo%20bar"); // true
687	 *
688	 * // Strict compare
689	 * isEqual("/foo", "foo", { leadingSlash: true }); // false
690	 * isEqual("foo/", "foo", { trailingSlash: true }); // false
691	 * isEqual("/foo bar", "/foo%20bar", { encoding: true }); // false
692	 * ```
693	 *
694	 * @group utils
695	 */
696	export function isEqual(a: string, b: string, options: CompareURLOptions = {}) {
697	  if (!options.trailingSlash) {
698	    a = withTrailingSlash(a);
699	    b = withTrailingSlash(b);
700	  }
701	  if (!options.leadingSlash) {
702	    a = withLeadingSlash(a);
703	    b = withLeadingSlash(b);
704	  }
705	  if (!options.encoding) {
706	    a = decode(a);
707	    b = decode(b);
708	  }
709	  return a === b;
710	}
